    20→function hashString(s: string): number {
    21→  let h = 0;
    22→  for (let i = 0; i < s.length; i++) h = s.charCodeAt(i) + ((h << 5) - h);
    23→  return Math.abs(h);
    24→}
    25→
    26→// Premium gradient covers — 6 distinct palettes, deterministic by name
    27→const COVER_PALETTES: { from: string; to: string; angle: number }[] = [
    28→  { from: "#2d1b3d", to: "#0e0e0e", angle: 135 },
    29→  { from: "#1b2d3a", to: "#0e0e0e", angle: 135 },
    30→  { from: "#3d2b1b", to: "#0e0e0e", angle: 135 },
    31→  { from: "#1b3a2d", to: "#0e0e0e", angle: 135 },
    32→  { from: "#3a1b2d", to: "#0e0e0e", angle: 135 },
    33→  { from: "#2d2d1b", to: "#0e0e0e", angle: 135 },
    34→];
    35→
    36→function gradientCover(name: string): string {
    37→  const h = hashString(name);
    38→  const p = COVER_PALETTES[h % COVER_PALETTES.length];
    39→  return `linear-gradient(${p.angle}deg, ${p.from}, ${p.to})`;
    40→}
    41→
    42→function patternStyle(name: string): React.CSSProperties {
    43→  const h = hashString(name);
    44→  const kind = h % 3;
    45→  if (kind === 0) {
    46→    return { backgroundImage: `radial-gradient(circle at 30% 30%, rgba(255,255,255,0.06) 0%, transparent 50%)` };
    47→  }
    48→  if (kind === 1) {
    49→    return { backgroundImage: `repeating-linear-gradient(45deg, transparent, transparent 12px, rgba(255,255,255,0.025) 12px, rgba(255,255,255,0.025) 24px)` };
    50→  }
    51→  return { backgroundImage: `radial-gradient(ellipse at 50% 0%, rgba(255,255,255,0.05) 0%, transparent 70%)` };
    52→}
    53→
    54→function formatTotalDuration(tracks: Track[]): string {
    55→  const totalSec = tracks.reduce((sum, t) => sum + (t.duration || 0), 0);
    56→  if (totalSec <= 0) return "";
    57→  const h = Math.floor(totalSec / 3600);
    58→  const m = Math.floor((totalSec % 3600) / 60);
    59→  if (h > 0) return `~${h} ч ${m} мин`;
    60→  return `~${m} мин`;
    61→}

  1417→            if (f) onCoverUpload(f);
  1418→            e.target.value = "";
  1419→          }}
  1420→        />
  1421→      </div>
  1422→
  1423→      {/* Title / meta */}
  1424→      {editing ? (
  1425→        <div className="flex items-center gap-1" onClick={(e) => e.stopPropagation()}>
  1426→          <input
  1427→            value={editName}
  1428→            onChange={(e) => onRenameChange(e.target.value)}
  1429→            className="flex-1 text-sm rounded-lg px-1.5 py-0.5 min-w-0 outline-none"
  1430→            style={{ backgroundColor: "var(--mq-input-bg)", border: "1px solid var(--mq-accent)", color: "var(--mq-text)" }}
  1431→            onKeyDown={(e) => {
  1432→              if (e.key === "Enter") onRenameConfirm();
  1433→              if (e.key === "Escape") onRenameCancel();
  1434→            }}
  1435→            autoFocus
  1436→          />
  1437→          <button onClick={onRenameConfirm} style={{ color: "var(--mq-accent)" }}>
  1438→            <Check className="w-3.5 h-3.5" />
  1439→          </button>
  1440→        </div>
  1441→      ) : (
  1442→        <>
  1443→          <p className="text-sm font-semibold truncate" style={{ color: "var(--mq-text)" }} title={pl.name}>
  1444→            {pl.name}
  1445→          </p>
  1446→          <div className="flex items-center gap-2 mt-1">
  1447→            <span className="mq-t-meta-2" style={{ color: "var(--mq-text-muted)" }}>
  1448→              {pl.tracks.length} треков
  1449→            </span>
  1450→            {totalDur && (
  1451→              <span className="mq-t-meta-2 flex items-center gap-0.5" style={{ color: "var(--mq-text-muted)", opacity: 0.6 }}>
  1452→                <Clock className="w-2.5 h-2.5" />
  1453→                {totalDur}
  1454→              </span>
  1455→            )}
  1456→          </div>
  1457→        </>
  1458→      )}
  1459→    </motion.div>
  1460→  );
  1461→}

  1478→function pluralRu(n: number, one: string, few: string, many: string): string {
  1479→  const mod10 = n % 10;
  1480→  const mod100 = n % 100;
  1481→  if (mod10 === 1 && mod100 !== 11) return one;
  1482→  if (mod10 >= 2 && mod10 <= 4 && (mod100 < 10 || mod100 >= 20)) return few;
  1483→  return many;
  1484→}
