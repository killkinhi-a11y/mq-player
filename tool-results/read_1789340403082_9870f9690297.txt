    65→/**
    66→ * WEB PARITY library — exact port of LibraryView.tsx (375px) with its lazy
    67→ * sub-views (FavoritesView / PlaylistView grid / HistoryView):
    68→ *
    69→ *  ┌ header: «Библиотека» 24/700 + «N элементов в коллекции»
    70→ *  ├ underline tab switcher: Избранное / Плейлисты / История (icon + label
    71→ *  │ + count badge, accent underline, 48dp rows)
    72→ *  ├ toolbar: «Поиск в библиотеке...» r12 + «Недавние» sort chip
    73→ *  ├ FAVORITES: icon chip + «Избранное» 18/700 + «N понр. · N не понр. ·
    74→ *  │ N подписок», pill tabs (icon + count), action bar («Найти трек...» +
    75→ *  │ sort + shuffle), «Слушать все»/«Перемешать» pills, r16 list card /
    76→ *  │ «Пока пусто» empty (72 r24 heart chip + «Искать музыку» CTA)
    77→ *  ├ PLAYLISTS: header + «Создать»/«Импорт», 2-col tile grid (r16 cards,
    78→ *  │ gradient art + ListMusic), mq-empty «Нет плейлистов», inline create
    79→ *  │ card («Новый плейлист», Отмена/Создать)
    80→ *  └ HISTORY: 48 r16 clock chip + «История» 20/700 + «Слушать всё»/clear,
    81→ *     search bar, 3+2 stat cards, grouped day cards («Сегодня»…), or
    82→ *     «История пуста» hero (96 r24 clock chip + «Начать слушать» CTA)
    83→ *
    84→ * LibraryBody is stateless (fixture-renderable for parity screenshots);
    85→ * LibraryScreen wires the live ViewModels into it.
    86→ */
    87→@Composable
    88→fun LibraryScreen(onOpenPlaylist: (String) -> Unit) {
    89→    val vm: PlaylistViewModel = viewModel()
    90→    val player: PlayerViewModel = viewModel()
    91→    val ui by vm.ui.collectAsState()
    92→    val queue by player.controller.queue.collectAsState()
    93→    val currentIndex by player.controller.currentIndex.collectAsState()
    94→    val favorites by player.favorites.collectAsState(initial = emptyList())
    95→    val history by ServiceLocator.localStore.history.collectAsState(initial = emptyList())
    96→
    97→    var tab by remember { mutableIntStateOf(0) }
    98→    var query by remember { mutableStateOf("") }
    99→    var sort by remember { mutableIntStateOf(0) }
   100→    var showCreate by remember { mutableStateOf(false) }
   101→    var newName by remember { mutableStateOf("") }
   102→
   103→    LaunchedEffect(Unit) { vm.refresh() }
   104→
   105→    LibraryBody(
   106→        ui = ui,
   107→        activeTab = tab,
   108→        favorites = favorites,
   109→        history = history,
   110→        playingTrackId = queue.getOrNull(currentIndex)?.id,
   111→        favoriteIds = favorites.map { it.id }.toSet(),
   112→        query = query,
   113→        sort = sort,
   114→        showCreate = showCreate,
   115→        createName = newName,
   116→        onTabChange = { tab = it; query = "" }, // web clears search on tab switch
   117→        onQueryChange = { query = it },
   118→        onSortChange = { sort = it },
   119→        onCreateToggle = { showCreate = !showCreate },
   120→        onCreateNameChange = { newName = it },
   121→        onCreateConfirm = { name ->
   122→            vm.create(name) { showCreate = false; newName = "" }
   123→        },
   124→        onOpenPlaylist = onOpenPlaylist,
   125→        onPlayQueue = { q, i -> player.controller.playQueue(q, i) },
   126→        onFavorite = { player.controller.toggleFavorite(it) },
   127→        onGoHome = { },
   128→    )
   129→}

   352→private fun LibraryTabBar(
   353→    activeTab: Int,
   354→    favCount: Int,
   355→    plCount: Int,
   356→    histCount: Int,
   357→    onTabChange: (Int) -> Unit,
   358→) {
   359→    val tabs = listOf(
   360→        Triple(MqIcons.Heart, "Избранное", favCount),

   366→            .padding(horizontal = 16.dp)
   367→            .fillMaxWidth()
   368→    ) {
   369→        Row(Modifier.fillMaxWidth()) {

   371→                val active = activeTab == i
   372→                Column(
   373→                    modifier = Modifier
   374→                        .weight(1f)
   375→                        .height(48.dp)
   376→                        .clickable { onTabChange(i) },

   378→                ) {

   380→                        Modifier.weight(1f).fillMaxWidth(),
   381→                        verticalAlignment = Alignment.CenterVertically,

   386→                            tint = if (active) MaterialTheme.colorScheme.primary
   387→                            else MaterialTheme.colorScheme.onSurfaceVariant,

   412→                    }
   413→                }

   418→            Modifier
   419→                .fillMaxWidth()
   420→                .height(2.dp)
   421→                .background(MaterialTheme.colorScheme.outline)

   428→                                Modifier
   429→                                    .fillMaxSize()
   430→                                    .padding(horizontal = 8.dp)
   431→                                    .background(
   432→                                        MaterialTheme.colorScheme.primary,
   433→                                        RoundedCornerShape(50)
   434→                                    )
   435→                            )
   436→                        }
   437→                    }
   438→                }
   439→            }
   440→        }
   441→    }
   442→}

  1366→/** N плейлист / плейлиста / плейлистов — web pluralRu. */
  1367→internal fun pluralPlaylists(n: Int): String = when {
  1368→    n % 10 == 1 && n % 100 != 11 -> "$n плейлист"
  1369→    n % 10 in 2..4 && n % 100 !in 12..14 -> "$n плейлиста"
  1370→    else -> "$n плейлистов"
  1371→}

  1373→/** web formatTotalDuration: 7854 → «2 ч 9 мин». */
  1374→internal fun formatTotal(seconds: Int): String {
  1375→    val h = seconds / 3600
  1376→    val m = (seconds % 3600) / 60
  1377→    return if (h > 0) "$h ч $m мин" else "$m мин"
  1378→}
