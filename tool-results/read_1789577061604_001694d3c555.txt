   154→function SettingRow({
   155→  icon: Icon, label, subtitle, value, onClick, danger, rightElement,
   156→}: {
   157→  icon: React.ElementType; label: string; subtitle?: string; value?: string;
   158→  onClick?: () => void; danger?: boolean; rightElement?: React.ReactNode;
   159→}) {
   160→  /* §HOVER/§PRESS (v71): plain button + CSS hover tint (one owner).
   161→     Was: Framer whileTap press-scale + whileHover bg — press feedback
   162→     removed app-wide; hover now matches SettingToggle (CSS only). */
   163→  const Wrapper: "button" | "div" = onClick ? "button" : "div";
   164→  return (
   165→    <Wrapper
   166→      {...(onClick ? { onClick } : {})}
   167→      className={`w-full flex items-center gap-3 px-3 sm:px-4 py-3 sm:py-3.5 text-left transition-colors hover:bg-[var(--mq-overlay-hover)] focus-visible:bg-[var(--mq-overlay-hover)]`}
   168→      style={{ borderTop: "1px solid var(--mq-border-hairline)" }}
   169→    >
   170→      <div
   171→        className="w-8 h-8 sm:w-9 sm:h-9 rounded-[var(--mq-r-card)] flex items-center justify-center flex-shrink-0"
   172→        style={{ backgroundColor: danger ? "rgba(239,68,68,0.08)" : "var(--mq-surface-2)", border: "1px solid " + (danger ? "rgba(239,68,68,0.15)" : "var(--mq-edge)") }}
   173→      >
   174→        <Icon className="w-4 h-4" style={{ color: danger ? "#ef4444" : "var(--mq-text-muted)" }} />
   175→      </div>
   176→      <div className="flex-1 min-w-0">
   177→        <p className="text-sm font-medium" style={{ color: danger ? "#ef4444" : "var(--mq-text)" }}>{label}</p>
   178→        {subtitle && <p className="mq-t-meta-2 sm:text-xs mt-0.5" style={{ color: "var(--mq-text-muted)" }}>{subtitle}</p>}
   179→      </div>
   180→      {value && (
   181→        <span className="text-xs px-2 py-1 rounded-full" style={{ backgroundColor: "var(--mq-glass-bg)", color: "var(--mq-text-muted)" }}>{value}</span>
   182→      )}
   183→      {rightElement}
   184→      {onClick && !rightElement && <ChevronRight className="w-4 h-4 flex-shrink-0" style={{ color: "var(--mq-text-muted)" }} />}
   185→    </Wrapper>
   186→  );
   187→}

   189→function SettingToggle({
   190→  icon: Icon, label, subtitle, value, onCheckedChange,
   191→}: {
   192→  icon: React.ElementType; label: string; subtitle?: string;
   193→  value: boolean; onCheckedChange: (v: boolean) => void;
   194→}) {
   195→  /* §HOVER one-owner: CSS owns background (hover:bg + transition-colors);
   196→     the old Framer whileHover={{ backgroundColor }} double-eased with the
   197→     CSS transition — one visible "highlight twice / second flash" bug. */
   198→  return (
   199→    <div
   200→      className="w-full flex items-center gap-3 px-3 sm:px-4 py-3 sm:py-3.5 transition-colors hover:bg-[var(--mq-overlay-hover)]"
   201→      style={{ borderTop: "1px solid var(--mq-border-hairline)" }}
   202→    >
   203→      <div
   204→        className="w-8 h-8 sm:w-9 sm:h-9 rounded-[var(--mq-r-card)] flex items-center justify-center flex-shrink-0"
   205→        style={{ backgroundColor: value ? "color-mix(in srgb, var(--mq-accent) 12%, transparent)" : "var(--mq-surface-2)", border: "1px solid " + (value ? "color-mix(in srgb, var(--mq-accent) 22%, transparent)" : "var(--mq-edge)") }}
   206→      >
   207→        <Icon className="w-4 h-4" style={{ color: value ? "var(--mq-accent)" : "var(--mq-text-muted)" }} />
   208→      </div>
   209→      <div className="flex-1 min-w-0">
   210→        <p className="text-sm font-medium" style={{ color: "var(--mq-text)" }}>{label}</p>
   211→        {subtitle && <p className="mq-t-meta-2 sm:text-xs mt-0.5" style={{ color: "var(--mq-text-muted)" }}>{subtitle}</p>}
   212→      </div>
   213→      <Switch checked={value} onCheckedChange={onCheckedChange} />
   214→    </div>
   215→  );
   216→}

   887→function urlBase64ToUint8Array(base64String: string): Uint8Array {
   888→  const padding = "=".repeat((4 - (base64String.length % 4)) % 4);
   889→  const base64 = (base64String + padding).replace(/-/g, "+").replace(/_/g, "/");
   890→  const rawData = typeof atob === "function" ? atob(base64) : Buffer.from(base64, "base64").toString("binary");
   891→  const buffer = new ArrayBuffer(rawData.length);
   892→  const output = new Uint8Array(buffer);
   893→  for (let i = 0; i < rawData.length; ++i) output[i] = rawData.charCodeAt(i);
   894→  return output;
   895→}
