   261→@Composable
   262→private fun AuthLandingCard(
   263→    onEmail: () -> Unit,
   264→    onRegister: () -> Unit,
   265→    onLoggedIn: (LocalStore.SessionUser) -> Unit,
   266→    onDemo: (LocalStore.SessionUser) -> Unit
   267→) {
   268→    val context = LocalContext.current
   269→    val scope = rememberCoroutineScope()
   270→    val auth = ServiceLocator.authRepository
   271→
   272→    var botName by remember { mutableStateOf<String?>(null) }
   273→    var botConfigured by remember { mutableStateOf<Boolean?>(null) }
   274→    var code by remember { mutableStateOf("") }
   275→    var username by remember { mutableStateOf("") }
   276→    var needsUsername by remember { mutableStateOf(false) }
   277→    var error by remember { mutableStateOf<String?>(null) }
   278→    var busy by remember { mutableStateOf(false) }
   279→    var demoBusy by remember { mutableStateOf(false) }
   280→    var googleBusy by remember { mutableStateOf(false) }
   281→
   282→    /**
   283→     * REAL Google native login — Credential Manager → backend bridge:
   284→     *   1. providers → PUBLIC web client id (the same one the backend verifies
   285→     *      the id_token audience against; the SECRET never enters the APK)
   286→     *   2. GET /api/auth/google/native → one-time nonce (HttpOnly cookie
   287→     *      captured by SecureCookieJar, replayed on the POST)
   288→     *   3. GetGoogleIdOption(serverClientId, nonce) → account picker →
   289→     *      Google-issued id_token (nonce claim bound)
   290→     *   4. POST /api/auth/google/native { idToken } → the SAME JWKS
   291→     *      verification + account resolution as the web callback →
   292→     *      Set-Cookie session → SecureCookieJar → Home
   293→     * Honest failures only: every error below is surfaced to the user —
   294→     * no catch-all swallowing, no fake navigation, no mock Home.
   295→     * Logs carry provider/step/status/class ONLY (never token material).
   296→     */
   297→    suspend fun googleNativeLogin() {
   298→        val clientId = auth.googleClientId()
   299→        if (clientId == null) {
   300→            error = "Google вход не настроен на сервере"
   301→            return
   302→        }
   303→        val nonce = auth.issueGoogleNativeNonce()
   304→        if (nonce == null) {
   305→            error = "Сеть недоступна. Проверьте подключение."
   306→            return
   307→        }
   308→        var idToken: String? = null
   309→        try {
   310→            val option = GetGoogleIdOption.Builder()
   311→                .setServerClientId(clientId)
   312→                .setNonce(nonce)
   313→                .build()
   314→            val request = GetCredentialRequest.Builder()
   315→                .addCredentialOption(option)
   316→                .build()
   317→            val credentialManager = CredentialManager.create(context)
   318→            val response = credentialManager.getCredential(context, request)
   319→            if (response.credential.type != GoogleIdTokenCredential.TYPE_GOOGLE_ID_TOKEN_CREDENTIAL) {
   320→                Log.w("MqAuth", "provider=google step=credential_manager status=unexpected_type")
   321→                error = "Не удалось получить Google-токен"
   322→                return
   323→            }
   324→            idToken = runCatching {
   325→                GoogleIdTokenCredential.createFrom(response.credential.data)
   326→            }.getOrNull()?.idToken
   327→        } catch (e: GetCredentialCancellationException) {
   328→            Log.d("MqAuth", "provider=google step=credential_manager status=cancelled")
   329→            return // user closed the account picker — not an error
   330→        } catch (e: NoCredentialException) {
   331→            Log.w("MqAuth", "provider=google step=credential_manager status=no_credentials class=${e.javaClass.simpleName}")
   332→            error = "Google-аккаунт не найден на устройстве. Добавьте аккаунт в настройках и повторите."
   333→            return
   334→        } catch (e: GetCredentialProviderConfigurationException) {
   335→            Log.w("MqAuth", "provider=google step=credential_manager status=provider_unavailable class=${e.javaClass.simpleName}")
   336→            error = "Google Play Services недоступен на устройстве"
   337→            return
   338→        } catch (e: GetCredentialException) {
   339→            Log.w("MqAuth", "provider=google step=credential_manager status=error class=${e.javaClass.simpleName}")
   340→            error = "Google вход не выполнен"
   341→            return
   342→        } catch (e: Exception) {
   343→            // device without Credential Manager support at all
   344→            Log.w("MqAuth", "provider=google step=credential_manager status=unavailable class=${e.javaClass.simpleName}")
   345→            error = "Google вход недоступен на этом устройстве"
   346→            return
   347→        }
   348→        if (idToken == null) {
   349→            error = "Не удалось получить Google-токен"
   350→            return
   351→        }
   352→        val result = auth.googleNativeLogin(idToken)
   353→        if (result == null) {
   354→            error = "Сеть недоступна. Проверьте подключение."
   355→            return
   356→        }
   357→        if (result.error != null || !result.authenticated || result.userId == null) {
   358→            error = when (result.error) {
   359→                "google_token_invalid" -> "Google-токен отклонён сервером"
   360→                "invalid_nonce" -> "Код входа устарел. Попробуйте ещё раз"
   361→                "blocked" -> "Аккаунт заблокирован"
   362→                "maintenance" -> "Технические работы. Вход временно недоступен"
   363→                "google_not_configured" -> "Google вход не настроен на сервере"
   364→                else -> "Не удалось войти через Google"
   365→            }
   366→            return
   367→        }
   368→        // Session cookie landed in SecureCookieJar (httpOnly Set-Cookie).
   369→        onLoggedIn(
   370→            LocalStore.SessionUser(
   371→                userId = result.userId ?: "",
   372→                username = result.username ?: "Google user",
   373→                role = result.role ?: "user",
   374→                avatar = result.avatar
   375→            )
   376→        )
   377→    }
   378→
   379→    LaunchedEffect(Unit) {
   380→        val name = auth.botName()
   381→        botConfigured = name != null
   382→        botName = name
   383→    }
   384→

   386→        // header

   388→            modifier = Modifier

   400→            color = MaterialTheme.colorScheme.onBackground,
   401→            modifier = Modifier.fillMaxWidth(), textAlign = TextAlign.Center
   402→        )
   403→        Text(
   404→            "Музыка, плейлисты и чаты синхронизированы между устройствами",
   405→            style = MqType.body.copy(fontSize = 14.sp, fontWeight = FontWeight.W400),
   406→            color = MaterialTheme.colorScheme.onSurfaceVariant,
   407→            modifier = Modifier.fillMaxWidth(), textAlign = TextAlign.Center
   408→        )
   409→        Spacer(Modifier.height(24.dp))
   410→
   411→        // Google — native Credential Manager login (the SAME backend flow the

   413→        WebButton(

   438→        Row(horizontalArrangement = Arrangement.Center, verticalAlignment = Alignment.CenterVertically) {

   444→        Text(
   445→            "Откройте бота в Telegram и отправьте любое сообщение. Бот пришлёт 6-значный код подтверждения",
   446→            style = MqType.meta.copy(fontSize = 12.sp),

   448→            modifier = Modifier.fillMaxWidth(), textAlign = TextAlign.Center

   466→            Spacer(Modifier.height(16.dp))

   468→
   469→        OtpBoxes(

   594→/** 6 single-digit boxes — web 44×56 r12, filled border telegram / error red. */
   595→@Composable
   596→private fun OtpBoxes(code: String, error: Boolean, onCode: (String) -> Unit) {
   597→    val inputBg = LocalMqPalette.current.inputBg
   598→    val outline = MaterialTheme.colorScheme.outline
   599→    Row(horizontalArrangement = Arrangement.spacedBy(10.dp)) {
   600→        repeat(6) { i ->
   601→            val ch = code.getOrNull(i)?.toString() ?: ""
   602→            val borderColor = when {
   603→                error -> Color(0xFFEF4444)
   604→                ch.isNotEmpty() -> Telegram
   605→                else -> outline
   606→            }
   607→            BasicTextField(
   608→                value = ch,
   609→                onValueChange = { v ->
   610→                    if (v.length <= 1 && (v.isEmpty() || v[0].isDigit())) {
   611→                        onCode(code.take(i) + v + code.drop(i + 1))
   612→                    }
   613→                },
   614→                keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.NumberPassword),
   615→                singleLine = true,
   616→                textStyle = MqType.display.copy(
   617→                    fontSize = 20.sp, fontWeight = FontWeight.W700,
   618→                    textAlign = TextAlign.Center
   619→                ).let { it.copy(color = MaterialTheme.colorScheme.onBackground) },
   620→                modifier = Modifier
   621→                    .width(44.dp)
   622→                    .height(56.dp)
   623→                    .clip(RoundedCornerShape(12.dp))
   624→                    .background(inputBg)
   625→                    .border(2.dp, borderColor, RoundedCornerShape(12.dp))
   626→            )
   627→        }
   628→    }
   629→}

   632→@Composable
   633→internal fun WebInput(
   634→    value: String,
   635→    onValue: (String) -> Unit,
   636→    placeholder: String,
   637→    password: Boolean = false,
   638→    keyboardType: KeyboardType = KeyboardType.Text
   639→) {

   642→        onValueChange = onValue,

   645→        keyboardOptions = KeyboardOptions(keyboardType = keyboardType),
   646→        textStyle = MqType.body.copy(color = MaterialTheme.colorScheme.onBackground),
   647→        decorationBox = { inner ->
   648→            Box(
   649→                Modifier
   650→                    .fillMaxWidth()
   651→                    .height(44.dp)
   652→                    .clip(RoundedCornerShape(12.dp))
   653→                    .background(LocalMqPalette.current.inputBg)
   654→                    .border(1.dp, MaterialTheme.colorScheme.outline, RoundedCornerShape(12.dp))
   655→                    .padding(horizontal = 14.dp),
   656→                contentAlignment = Alignment.CenterStart
   657→            ) {
   658→                if (value.isEmpty()) {
   659→                    Text(placeholder, style = MqType.body.copy(fontSize = 14.sp), color = MaterialTheme.colorScheme.onSurfaceVariant)
   660→                }
   661→                inner()
   662→            }
   663→        }
   664→    )
