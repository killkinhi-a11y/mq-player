    87→    fun onQueryChange(query: String) {

    98→                _ui.value = _ui.value.copy(error = "Поиск недоступен, проверьте сеть")
    99→                emptyList()
   100→            }
   101→            _ui.value = _ui.value.copy(
   102→                results = results, loading = false, searched = true, error = null
   103→            )
   104→        }
   105→    }

   227→            }
   228→        }
   229→    }

   231→    fun next() {
   232→        viewModelScope.launch {
   233→            _ui.value = _ui.value.copy(loading = true)

   238→    }

   287→class ChatDetailViewModel : ViewModel() {
   288→    data class ChatDetailUi(
   289→        val peerId: String = "",
   290→        val peerName: String = "",
   291→        val peerAvatar: String? = null,
   292→        val messages: List<com.mq1.player.data.api.MessageDto> = emptyList(),
   293→        val loading: Boolean = true,
   294→        val error: String? = null
   295→    )
   296→
   297→    private val social = ServiceLocator.socialRepository
   298→    private val hub = ServiceLocator.socialHub
   299→    private val _ui = MutableStateFlow(ChatDetailUi())
   300→    val ui: StateFlow<ChatDetailUi> = _ui
   301→    private var pollJob: Job? = null
   302→
   303→    fun load(peerId: String, peerName: String) {
   304→        _ui.value = ChatDetailUi(
   305→            peerId = peerId,
   306→            peerName = peerName,
   307→            peerAvatar = hub.state.value.friends.firstOrNull { it.id == peerId }?.avatar
   308→        )
   309→        // F7: opening the chat clears that peer's unread badge
   310→        hub.markPeerRead(peerId)
   311→        fetch()
   312→        pollJob?.cancel()
   313→        pollJob = viewModelScope.launch {
   314→            while (true) {
   315→                delay(5000)
   316→                fetch()
   317→            }
   318→        }
   319→    }
   320→
   321→    private fun fetch() {
   322→        viewModelScope.launch {
   323→            val messages = social.messages(_ui.value.peerId)
   324→            if (messages.isEmpty() && _ui.value.messages.isEmpty() && _ui.value.loading) {
   325→                // Distinguish "no messages yet" from a failed fetch: poll keeps
   326→                // retrying silently either way (5s cadence, real data only).
   327→                _ui.value = _ui.value.copy(loading = false)
   328→            } else {
   329→                _ui.value = _ui.value.copy(messages = messages, loading = false)
   330→                hub.markPeerRead(_ui.value.peerId)
   331→            }
   332→        }
   333→    }
   334→
   335→    fun send(text: String) {
   336→        val peerId = _ui.value.peerId
   337→        if (text.isBlank() || peerId.isBlank()) return
   338→        viewModelScope.launch {
   339→            val sent = social.send(peerId, text.trim())
   340→            if (sent != null) fetch()
   341→            else _ui.value = _ui.value.copy(error = "Сообщение не отправлено")
   342→        }
   343→    }
   344→
   345→    override fun onCleared() {
   346→        pollJob?.cancel()
   347→        super.onCleared()
   348→    }
   349→}

   359→        val query: String = "",

   361→        /** Initial sync in flight (first poll after foreground/login). */
   362→        val loading: Boolean = true,
   363→        /** Last sync failed and we have no data yet → full-screen error+retry. */
   364→        val error: Boolean = false,
   365→        /** Result snackbar (action outcomes). */
   366→        val message: String? = null,
   367→        /** Row ids with an in-flight action (disable buttons, show progress). */
   368→        val busyIds: Set<String> = emptySet()
   369→    )
   370→
   371→    private val social = ServiceLocator.socialRepository
   372→    private val hub = ServiceLocator.socialHub
   373→    private val _ui = MutableStateFlow(FriendsUi())
   374→    val ui: StateFlow<FriendsUi> = _ui
   375→    private var searchJob: Job? = null
   376→
   377→    init {
   378→        viewModelScope.launch {
   379→            hub.state.collect { s ->
   380→                _ui.value = _ui.value.copy(
   381→                    friends = s.friends,
   382→                    incoming = s.incoming,
   383→                    outgoing = s.outgoing,
   384→                    online = s.online,
   385→                    unreadCounts = s.unreadCounts,
   386→                    loading = !s.synced,
   387→                    error = s.syncError && !s.synced
   388→                )
   389→            }
   390→        }
   391→    }
   392→
   393→    fun retry() = hub.refreshNow()
   394→
   395→    fun search(query: String) {
   396→        _ui.value = _ui.value.copy(query = query)
   397→        searchJob?.cancel()
   398→        if (query.length < 2) {
   399→            _ui.value = _ui.value.copy(found = emptyList(), searching = false)
   400→            return
   401→        }
   402→        searchJob = viewModelScope.launch {
   403→            delay(300) // debounce — same as web
   404→            _ui.value = _ui.value.copy(searching = true)
   405→            val results = social.searchUsers(query)
   406→            // Only current query's results survive (stale-response guard)
   407→            if (query == _ui.value.query) {
   408→                _ui.value = _ui.value.copy(found = results, searching = false)
   409→            }
   410→        }
   411→    }
   412→
   413→    private fun runAction(rowKey: String, action: suspend () -> Result<Unit>, onSuccess: String) {
   414→        if (rowKey in _ui.value.busyIds) return
   415→        _ui.value = _ui.value.copy(busyIds = _ui.value.busyIds + rowKey)
   416→        viewModelScope.launch {
   417→            action()
   418→                .onSuccess {
   419→                    _ui.value = _ui.value.copy(message = onSuccess)
   420→                    hub.refreshNow()
   421→                }
   422→                .onFailure { _ui.value = _ui.value.copy(message = it.message ?: "Ошибка") }
   423→            _ui.value = _ui.value.copy(busyIds = _ui.value.busyIds - rowKey)
   424→        }
   425→    }

   568→ */
   569→class LyricsViewModel : ViewModel() {
   570→
   571→    data class LyricsUi(
   572→        val trackKey: String = "",
   573→        val lines: List<com.mq1.player.data.api.LyricLine> = emptyList(),
   574→        val plainText: String = "",
   575→        val synced: Boolean = false,
   576→        val loading: Boolean = false,
   577→        /** true when the fetch completed but nothing was found. */
   578→        val unavailable: Boolean = false
   579→    )
   580→
   581→    private val music = ServiceLocator.musicRepository
   582→    private val _ui = MutableStateFlow(LyricsUi())
   583→    val ui: StateFlow<LyricsUi> = _ui
   584→
   585→    /** Load (once per track) when the lyrics panel opens. */
   586→    fun loadIfNeeded(track: com.mq1.player.data.api.Track) {
   587→        val key = track.scTrackId?.toString() ?: track.id
   588→        if (_ui.value.trackKey == key && (_ui.value.loading || _ui.value.synced ||
   589→                _ui.value.plainText.isNotBlank() || _ui.value.unavailable)) return
   590→        _ui.value = LyricsUi(trackKey = key, loading = true)
   591→        viewModelScope.launch {
   592→            val result = music.lyrics(track.artist, track.title)
   593→            _ui.value = if (result == null) {
   594→                _ui.value.copy(loading = false, unavailable = true)
   595→            } else if (result.synced && result.lyrics.isNotEmpty()) {
   596→                // Server-side LRC parse occasionally leaves an embedded
   597→                // [mm:ss.xx] prefix in the text — strip it for display.
   598→                val embeddedTs = Regex("^\\[\\d{2}:\\d{2}(?:[.:]\\d{2,3})?]\\s*")
   599→                _ui.value.copy(
   600→                    loading = false,
   601→                    lines = result.lyrics.map { it.copy(text = it.text.replace(embeddedTs, "")) },
   602→                    synced = true
   603→                )
   604→            } else if (result.plainText.isNotBlank()) {
   605→                _ui.value.copy(loading = false, plainText = result.plainText)
   606→            } else {
   607→                _ui.value.copy(loading = false, unavailable = true)
   608→            }
   609→        }
   610→    }
   611→}
   612→
   613→// ─────────────────────────────────────────────────────────────────────────────
   614→// F9 — own profile: account (server) + local content + friends (hub) + editing
   615→// ─────────────────────────────────────────────────────────────────────────────
   616→
   617→/**
   618→ * Own profile. Data sources (all REAL, no invented state):
   619→ *  - GET /api/user/profile   → account: username, email, avatar, role, createdAt
   620→ *  - GET /api/auth/me        → account state: confirmed, telegramUsername
   621→ *  - GET /api/playlists?myOnly=true → my playlists
   622→ *  - LocalStore favorites    → likes (web parity: per-device, documented)
   623→ *  - LocalStore history      → recent activity
   624→ *  - SocialHub               → friends + online (shared F7 state)
   625→ * Editing: POST /api/user/avatar + POST /api/auth/update-username (web contract);
   626→ * the session cache is updated so every surface sees the new identity.
   627→ */
   628→class MyProfileViewModel : ViewModel() {
   629→
   630→    data class ProfileUi(
   631→        // ── Account (server) ──
   632→        val userId: String = "",
   633→        val username: String = "",
   634→        val email: String? = null,
   635→        val avatar: String? = null,
   636→        val role: String = "user",
   637→        val createdAt: String? = null,
   638→        val telegramUsername: String? = null,
   639→        val confirmed: Boolean? = null,
   640→        // ── Content ──
   641→        val playlists: List<com.mq1.player.data.api.PlaylistDto> = emptyList(),
   642→        val likes: List<Track> = emptyList(),
   643→        val history: List<Track> = emptyList(),
   644→        val friends: List<com.mq1.player.data.api.Friend> = emptyList(),
   645→        val online: Map<String, Boolean> = emptyMap(),
   646→        // ── States ──
   647→        val loading: Boolean = true,
   648→        val error: String? = null,
   649→        val savingUsername: Boolean = false,
   650→        val savingAvatar: Boolean = false,
   651→        /** Snackbar result messages (edit outcomes). */
   652→        val message: String? = null
   653→    ) {
   654→        /** Top artists derived from likes — tap → artist screen. */
   655→        val topArtists: List<String>
   656→            get() = likes.groupBy { it.artist }
   657→                .filterKeys { it.isNotBlank() }
   658→                .map { (artist, tracks) -> artist to tracks.size }
   659→                .sortedByDescending { it.second }
   660→                .take(8)
   661→                .map { it.first }
   662→    }
   663→
   664→    private val profileRepo = ServiceLocator.profileRepository
   665→    private val playlistsRepo = ServiceLocator.playlistRepository
   666→    private val local = ServiceLocator.localStore
   667→    private val hub = ServiceLocator.socialHub
   668→
   669→    private val _ui = MutableStateFlow(ProfileUi())
   670→    val ui: StateFlow<ProfileUi> = _ui
   671→
   672→    init {
   673→        // Local content (favorites/history) and social state — live flows.
   674→        viewModelScope.launch {
   675→            local.favorites.collect { likes ->
   676→                _ui.value = _ui.value.copy(likes = likes)
   677→            }
   678→        }
   679→        viewModelScope.launch {
   680→            local.history.collect { history ->
   681→                _ui.value = _ui.value.copy(history = history.take(20))
   682→            }
   683→        }
   684→        viewModelScope.launch {
   685→            hub.state.collect { s ->
   686→                _ui.value = _ui.value.copy(friends = s.friends, online = s.online)
   687→            }
   688→        }
   689→        refresh()
   690→    }
   691→
   692→    fun refresh() {
   693→        viewModelScope.launch {
   694→            _ui.value = _ui.value.copy(loading = true, error = null)
   695→
   696→            val account = profileRepo.myProfile()
   697→            if (account.isFailure) {
   698→                _ui.value = _ui.value.copy(
   699→                    loading = false,
   700→                    error = account.exceptionOrNull()?.message ?: "Профиль недоступен"
   701→                )
   702→                return@launch
   703→            }
   704→            val me = runCatching {
   705→                ServiceLocator.api.me().body()
   706→            }.getOrNull()
   707→
   708→            val myPlaylists = runCatching { playlistsRepo.myPlaylists() }.getOrElse { emptyList() }
   709→
   710→            _ui.value = _ui.value.copy(
   711→                userId = account.getOrThrow().id,
   712→                username = account.getOrThrow().username,
   713→                email = account.getOrThrow().email,
   714→                avatar = account.getOrThrow().avatar,
   715→                role = account.getOrThrow().role,
   716→                createdAt = account.getOrThrow().createdAt,
   717→                telegramUsername = me?.telegramUsername,
   718→                confirmed = me?.confirmed,
   719→                playlists = myPlaylists,
   720→                loading = false
   721→            )
   722→        }
   723→    }
   724→
   725→    /** Username availability (used with debounce by the edit dialog). */
   726→    suspend fun checkUsername(name: String): com.mq1.player.data.api.UsernameCheckResponse? =
   727→        profileRepo.checkUsername(name, _ui.value.userId).getOrNull()
   728→
   729→    fun saveUsername(newUsername: String) {
   730→        if (_ui.value.savingUsername) return
   731→        val localError = com.mq1.player.data.repo.ProfileRepository.validateUsername(newUsername)
   732→        if (localError != null || newUsername == _ui.value.username) {
   733→            _ui.value = _ui.value.copy(message = localError ?: "Имя не изменилось")
   734→            return
   735→        }
   736→        viewModelScope.launch {
   737→            _ui.value = _ui.value.copy(savingUsername = true)
   738→            profileRepo.updateUsername(newUsername)
   739→                .onSuccess { response ->
   740→                    _ui.value = _ui.value.copy(
   741→                        username = response.username ?: newUsername,
   742→                        savingUsername = false,
   743→                        message = response.message ?: "Имя обновлено"
   744→                    )
   745→                    // Session cache → all surfaces (Settings, chats, hub gate).
   746→                    local.sessionUser.firstOrNull()?.let { cached ->
   747→                        local.setSessionUser(cached.copy(username = response.username ?: newUsername))
   748→                    }
   749→                }
   750→                .onFailure {
   751→                    _ui.value = _ui.value.copy(
   752→                        savingUsername = false,
   753→                        message = it.message ?: "Ошибка сохранения имени"
   754→                    )
   755→                }
   756→        }
   757→    }
   758→
   759→    fun saveAvatar(imageBytes: ByteArray) {
   760→        if (_ui.value.savingAvatar) return
   761→        viewModelScope.launch(kotlinx.coroutines.Dispatchers.IO) {
   762→            val dataUrl = com.mq1.player.data.repo.ProfileRepository.bitmapToAvatarDataUrl(imageBytes)
   763→            if (dataUrl == null) {
   764→                _ui.value = _ui.value.copy(message = "Не удалось прочитать изображение")
   765→                return@launch
   766→            }
   767→            _ui.value = _ui.value.copy(savingAvatar = true, avatar = dataUrl) // optimistic
   768→            profileRepo.uploadAvatar(dataUrl)
   769→                .onSuccess { response ->
   770→                    _ui.value = _ui.value.copy(
   771→                        avatar = response.avatar ?: dataUrl,
   772→                        savingAvatar = false,
   773→                        message = response.message ?: "Аватарка обновлена"
   774→                    )
   775→                    local.sessionUser.firstOrNull()?.let { cached ->
   776→                        local.setSessionUser(cached.copy(avatar = dataUrl))
   777→                    }
   778→                }
   779→                .onFailure {
   780→                    _ui.value = _ui.value.copy(
   781→                        // rollback to the server value we still have in profile
   782→                        avatar = runCatching {
   783→                            profileRepo.myProfile().getOrNull()?.avatar
   784→                        }.getOrNull() ?: _ui.value.avatar,
   785→                        savingAvatar = false,
   786→                        message = it.message ?: "Ошибка загрузки аватара"
   787→                    )
   788→                }
   789→        }
   790→    }
   791→
   792→    fun consumeMessage() {
   793→        _ui.value = _ui.value.copy(message = null)
   794→    }
   795→}
