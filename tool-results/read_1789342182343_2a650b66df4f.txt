    64→/**
    65→ * WEB PARITY search — exact port of SearchView.tsx mobile (375px) composition:
    66→ *
    67→ *  ┌ header: «Поиск» mq-t-display 26/800
    68→ *  ├ search bar: h48 r14 surface bg + hairline border, Search 18 left,
    69→ *  │ placeholder «Искать треки, артисты, альбомы...», X clear 28 circle
    70→ *  ├ mobile action row (right-aligned): «Фильтры» + «Файлы» h44 r12
    71→ *  ├ genre chips (collapsible): «Все» + 8 genres, pill r-full 12/600
    72→ *  ├ recent queries: «Недавние запросы» + «Очистить», clock chips r12
    73→ *  ├ results head: «Результаты» 20/700 + count + «Играть все» accent pill
    74→ *  ├ loading: 5 skeleton rows (art 44 + 2 lines + duration)
    75→ *  ├ no-results: mq-empty dashed card + 4 trending retry chips
    76→ *  ├ results: shared TrackRow list
    77→ *  └ idle hero: 88×88 r28 gradient Headphones chip, «Что послушаем?»,
    78→ *     «Популярные запросы» + 7 trending chips (Поп…Джаз)
    79→ *
    80→ * SearchBody is stateless (fixture-renderable for parity screenshots);
    81→ * SearchScreen wires the live ViewModels into it.
    82→ */
    83→@Composable
    84→fun SearchScreen(onOpenArtist: (String) -> Unit) {
    85→    val vm: SearchViewModel = viewModel()
    86→    val player: PlayerViewModel = viewModel()
    87→    val ui by vm.ui.collectAsState()
    88→    val queue by player.controller.queue.collectAsState()
    89→    val currentIndex by player.controller.currentIndex.collectAsState()
    90→    val favorites by player.favorites.collectAsState(initial = emptyList())
    91→
    92→    SearchBody(
    93→        ui = ui,
    94→        playingTrackId = queue.getOrNull(currentIndex)?.id,
    95→        favorites = favorites,
    96→        onQueryChange = vm::onQueryChange,
    97→        onClearHistory = vm::clearHistory,
    98→        onRemoveHistoryItem = vm::removeHistoryItem,
    99→        onPlayQueue = { q, i -> player.controller.playQueue(q, i) },
   100→        onFavorite = { player.controller.toggleFavorite(it) },
   101→        onOpenArtist = onOpenArtist,
   102→        onRetry = { vm.onQueryChange(ui.query) },
   103→    )
   104→}

   522→                                    .background(MaterialTheme.colorScheme.surfaceContainer)
   523→                                    .border(1.dp, MaterialTheme.colorScheme.outline.copy(alpha = 0.36f), RoundedCornerShape(50))
   524→                                    .clickable { onQueryChange(term) }
   525→                                    .padding(horizontal = 14.dp, vertical = 6.dp)
   526→                            )
   527→                        }
   528→                    }
   529→                }
   530→            }
   531→        }
   532→
   533→        // ── results (shared TrackRow) ───────────────────────────────────
   534→        if (!ui.loading && ui.results.isNotEmpty()) {
   535→            item {
   536→                Text(
   537→                    "ТРЕКИ · ${ui.results.size}",
   538→                    style = MqType.track.copy(fontWeight = FontWeight.Bold, letterSpacing = 0.7.sp),
   539→                    color = MaterialTheme.colorScheme.onSurfaceVariant,
   540→                    modifier = Modifier.padding(horizontal = 20.dp, vertical = 4.dp)
   541→                )
   542→            }
   543→            items(ui.results, key = { it.id }) { track ->
   544→                TrackRow(
   545→                    track = track,
   546→                    isPlaying = playingTrackId == track.id,
   547→                    isFavorite = track.id in favoriteIds,
   548→                    onPlay = { onPlayQueue(ui.results, ui.results.indexOf(track)) },
   549→                    onFavorite = { onFavorite(track) },
   550→                    onMenu = { },
   551→                    modifier = Modifier.padding(horizontal = 8.dp)
   552→                )
   553→            }
   554→        }
   555→
   556→        // ── error state ─────────────────────────────────────────────────
   557→        if (ui.error != null) {
   558→            item { ErrorState(ui.error!!, onRetry = onRetry) }
   559→        }
   560→
   561→        // ── idle hero (web default empty state) ─────────────────────────
   562→        if (!ui.searched && !ui.loading && ui.error == null && ui.history.isEmpty()) {
   563→            item { SearchIdleHero(onQueryChange = onQueryChange) }
   564→        }
   565→
   566→        // ── trending chips (web shows these when history exists) ────────
   567→        if (!ui.searched && !ui.loading && ui.error == null && ui.history.isNotEmpty()) {
   568→            item { TrendingSection(onQueryChange = onQueryChange) }
   569→        }
   570→    }
   571→}
