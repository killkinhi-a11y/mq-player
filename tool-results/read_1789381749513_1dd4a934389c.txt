   477→/** 6 single-digit boxes — web 44×56 r12, filled border telegram / error red. */
   478→@Composable
   479→private fun OtpBoxes(code: String, error: Boolean, onCode: (String) -> Unit) {
   480→    val inputBg = LocalMqPalette.current.inputBg
   481→    val outline = MaterialTheme.colorScheme.outline
   482→    Row(horizontalArrangement = Arrangement.spacedBy(10.dp)) {
   483→        repeat(6) { i ->
   484→            val ch = code.getOrNull(i)?.toString() ?: ""
   485→            val borderColor = when {
   486→                error -> Color(0xFFEF4444)
   487→                ch.isNotEmpty() -> Telegram
   488→                else -> outline
   489→            }
   490→            BasicTextField(
   491→                value = ch,
   492→                onValueChange = { v ->
   493→                    if (v.length <= 1 && (v.isEmpty() || v[0].isDigit())) {
   494→                        onCode(code.take(i) + v + code.drop(i + 1))
   495→                    }
   496→                },

   550→// ── email login card (web authStep "login") ───────────────────────────────
   551→
   552→@Composable
   553→private fun EmailLoginCard(
   554→    onBack: () -> Unit,
   555→    onLoggedIn: (LocalStore.SessionUser) -> Unit,
   556→    onNeedConfirm: () -> Unit
   557→) {
   558→    val scope = rememberCoroutineScope()
   559→    val auth = ServiceLocator.authRepository
   560→    var email by remember { mutableStateOf("") }
   561→    var password by remember { mutableStateOf("") }
   562→    var error by remember { mutableStateOf<String?>(null) }
   563→    var busy by remember { mutableStateOf(false) }
   564→
   565→    AuthCard {
   566→        BackHeader("Вход по email", onBack)
   567→        Spacer(Modifier.height(16.dp))
   568→        WebInput(email, { email = it }, "email@example.com", keyboardType = KeyboardType.Email)
   569→        Spacer(Modifier.height(12.dp))
   570→        WebInput(password, { password = it }, "Пароль", password = true)
   571→        error?.let {
   572→            Spacer(Modifier.height(12.dp))
   573→            Text(it, style = MqType.body, color = Color(0xFFFF6B6B), modifier = Modifier.fillMaxWidth(), textAlign = TextAlign.Center)
   574→            if (it.contains("подтверд", ignoreCase = true)) {
   575→                Spacer(Modifier.height(8.dp))
   576→                Text(
   577→                    "Подтвердить почту", style = MqType.body.copy(fontWeight = FontWeight.W500),
   578→                    color = MaterialTheme.colorScheme.primary,
   579→                    modifier = Modifier
   580→                        .align(Alignment.CenterHorizontally)
   581→                        .clickable { onNeedConfirm() }
   582→                )
   583→            }
   584→        }
   585→        Spacer(Modifier.height(16.dp))
   586→        WebButton(
   587→            text = "Войти", bg = MaterialTheme.colorScheme.primary,
   588→            border = MaterialTheme.colorScheme.primary, textColor = MaterialTheme.colorScheme.onPrimary,
   589→            enabled = email.isNotBlank() && password.isNotBlank() && !busy,
   590→            loading = busy
   591→        ) {
   592→            busy = true; error = null
   593→            scope.launch {
   594→                val result = auth.loginWithEmail(email, password)
   595→                busy = false
   596→                when {
   597→                    result == null -> error = "Ошибка соединения. Проверьте интернет."
   598→                    result.userId != null -> onLoggedIn(
   599→                        LocalStore.SessionUser(
   600→                            userId = result.userId ?: "", username = result.username ?: "",
   601→                            role = result.role ?: "user", avatar = result.avatar
   602→                        )
   603→                    )
   604→                    else -> error = result.error ?: result.message ?: "Ошибка входа"
   605→                }
   606→            }
   607→        }
   608→    }
   609→}

   611→// ── register card (web authStep "register") ───────────────────────────────
   612→
   613→@Composable
   614→private fun RegisterCard(
   615→    email: String,
   616→    onEmail: (String) -> Unit,
   617→    onBack: () -> Unit,
   618→    onRegistered: () -> Unit
   619→) {
   620→    val scope = rememberCoroutineScope()
   621→    val auth = ServiceLocator.authRepository
   622→    var username by remember { mutableStateOf("") }
   623→    var email by remember { mutableStateOf("") }
   624→    var password by remember { mutableStateOf("") }
   625→    var error by remember { mutableStateOf<String?>(null) }
   626→    var busy by remember { mutableStateOf(false) }
   627→
   628→    AuthCard {
   629→        BackHeader("Регистрация", onBack)
   630→        Spacer(Modifier.height(16.dp))
   631→        WebInput(username, { username = it }, "Имя пользователя")
   632→        Spacer(Modifier.height(12.dp))
   633→        WebInput(email, { onEmail(it) }, "email@example.com", keyboardType = KeyboardType.Email)
   634→        Spacer(Modifier.height(12.dp))
   635→        WebInput(password, { password = it }, "Пароль (мин. 6 символов)", password = true)
   636→        error?.let {
   637→            Spacer(Modifier.height(12.dp))
   638→            Text(it, style = MqType.body, color = Color(0xFFFF6B6B), modifier = Modifier.fillMaxWidth(), textAlign = TextAlign.Center)
   639→        }
   640→        Spacer(Modifier.height(16.dp))
   641→        WebButton(
   642→            text = "Создать аккаунт", bg = MaterialTheme.colorScheme.primary,
   643→            border = MaterialTheme.colorScheme.primary, textColor = MaterialTheme.colorScheme.onPrimary,
   644→            enabled = username.isNotBlank() && email.isNotBlank() && password.length >= 6 && !busy,
   645→            loading = busy
   646→        ) {
   647→            busy = true; error = null
   648→            scope.launch {
   649→                val result = auth.registerEmail(username, email, password)
   650→                busy = false
   651→                if (result == null) {
   652→                    error = "Ошибка соединения. Проверьте интернет."
   653→                } else if (result.devCode != null || result.error == null) {
   654→                    onRegistered()
   655→                } else {
   656→                    error = result.error ?: "Ошибка регистрации"
   657→                }
   658→            }
   659→        }
   660→    }
   661→}

   665→@Composable
   666→private fun ConfirmCard(
   667→    email: String,
   668→    onConfirmed: (LocalStore.SessionUser) -> Unit,
   669→    onBack: () -> Unit
   670→) {
   671→    val scope = rememberCoroutineScope()
   672→    val auth = ServiceLocator.authRepository
   673→    var code by remember { mutableStateOf("") }
   674→    var error by remember { mutableStateOf<String?>(null) }
   675→    var busy by remember { mutableStateOf(false) }
   676→
   677→    AuthCard {
   678→        BackHeader("Подтвердите почту", onBack)
   679→        Spacer(Modifier.height(16.dp))
   680→        Text(
   681→            "Введите 6-значный код из письма",
   682→            style = MqType.body, color = MaterialTheme.colorScheme.onSurfaceVariant,
   683→            modifier = Modifier.fillMaxWidth(), textAlign = TextAlign.Center
   684→        )
   685→        Spacer(Modifier.height(16.dp))
   686→        Row(Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.Center) {
   687→            OtpBoxes(code, error != null) { code = it }
   688→        }
   689→        error?.let {
   690→            Spacer(Modifier.height(12.dp))
   691→            Text(it, style = MqType.body, color = Color(0xFFFF6B6B), modifier = Modifier.fillMaxWidth(), textAlign = TextAlign.Center)

   696→            enabled = code.length == 6 && !busy, loading = busy,
   697→            modifier = Modifier.fillMaxWidth()
   698→        ) {
   699→            busy = true; error = null
   700→            scope.launch {
   701→                val result = if (email.isBlank()) null else auth.confirmEmailCode(email, code)
   702→                busy = false
   703→                when {
   704→                    result == null -> error = "Не указан email. Начните регистрацию заново."

   706→                        LocalStore.SessionUser(
   707→                            userId = result.userId ?: "", username = result.username ?: "",
   708→                            role = result.role ?: "user", avatar = result.avatar
   709→                        )
   710→                    )
   711→                    else -> error = result.error ?: result.message ?: "Неверный код"
   712→                }
   713→            }
   714→        }
   715→    }
   716→}
