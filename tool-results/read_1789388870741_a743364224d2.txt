    73→class SearchViewModel : ViewModel() {
    74→    data class SearchUi(
    75→        val query: String = "",
    76→        val results: List<Track> = emptyList(),
    77→        val loading: Boolean = false,
    78→        val searched: Boolean = false,
    79→        val error: String? = null,
    80→        // web SEARCH_HISTORY_KEY — recent queries for the «Недавние запросы»
    81→        // chips (in-memory parity of the web localStorage list, max 15)
    82→        val history: List<String> = emptyList()
    83→    )
    84→
    85→    private val music = ServiceLocator.musicRepository
    86→    private val _ui = MutableStateFlow(SearchUi())
    87→    val ui: StateFlow<SearchUi> = _ui
    88→    private var searchJob: Job? = null
    89→
    90→    fun onQueryChange(query: String) {
    91→        _ui.value = _ui.value.copy(query = query)
    92→        searchJob?.cancel()
    93→        if (query.isBlank()) {
    94→            _ui.value = _ui.value.copy(results = emptyList(), searched = false, loading = false)
    95→            return
    96→        }
    97→        searchJob = viewModelScope.launch {
    98→            delay(300) // debounce — dedupes typing bursts (P20.7)
    99→            _ui.value = _ui.value.copy(loading = true)
   100→            val results = runCatching { music.search(query) }.getOrElse {
   101→                _ui.value = _ui.value.copy(error = "Поиск недоступен, проверьте сеть")
   102→                emptyList()
   103→            }
   104→            val trimmed = query.trim()
   105→            val history = if (results.isNotEmpty() && trimmed.isNotBlank()) {
   106→                listOf(trimmed) + _ui.value.history.filter {
   107→                    it.lowercase() != trimmed.lowercase()
   108→                }.take(14)
   109→            } else _ui.value.history
   110→            _ui.value = _ui.value.copy(
   111→                results = results, loading = false, searched = true,
   112→                error = null, history = history
   113→            )
   114→        }
   115→    }
   116→
   117→    /** Web handleClearHistory — clears the recent-query chips. */
   118→    fun clearHistory() {
   119→        _ui.value = _ui.value.copy(history = emptyList())
   120→    }
   121→
   122→    /** Web handleRemoveHistoryItem — drops one chip. */
   123→    fun removeHistoryItem(query: String) {
   124→        _ui.value = _ui.value.copy(
   125→            history = _ui.value.history.filter { it.lowercase() != query.lowercase() }
   126→        )
   127→    }
   128→}

   345→class ChatDetailViewModel : ViewModel() {
   346→    data class ChatDetailUi(
   347→        val peerId: String = "",
   348→        val peerName: String = "",
   349→        val peerAvatar: String? = null,
   350→        val messages: List<com.mq1.player.data.api.MessageDto> = emptyList(),
   351→        val loading: Boolean = true,
   352→        val error: String? = null
   353→    )
   354→
   355→    private val social = ServiceLocator.socialRepository
   356→    private val hub = ServiceLocator.socialHub
   357→    private val _ui = MutableStateFlow(ChatDetailUi())
   358→    val ui: StateFlow<ChatDetailUi> = _ui
   359→    private var pollJob: Job? = null
   360→
   361→    fun load(peerId: String, peerName: String) {
   362→        _ui.value = ChatDetailUi(
   363→            peerId = peerId,
   364→            peerName = peerName,
   365→            peerAvatar = hub.state.value.friends.firstOrNull { it.id == peerId }?.avatar
   366→        )
   367→        // F7: opening the chat clears that peer's unread badge
   368→        hub.markPeerRead(peerId)
   369→        fetch()
   370→        pollJob?.cancel()
   371→        pollJob = viewModelScope.launch {
   372→            while (true) {
   373→                delay(5000)
   374→                fetch()
   375→            }
   376→        }
   377→    }
   378→
   379→    private fun fetch() {
   380→        viewModelScope.launch {
   381→            val messages = social.messages(_ui.value.peerId)
   382→            if (messages.isEmpty() && _ui.value.messages.isEmpty() && _ui.value.loading) {
   383→                // Distinguish "no messages yet" from a failed fetch: poll keeps
   384→                // retrying silently either way (5s cadence, real data only).
   385→                _ui.value = _ui.value.copy(loading = false)
   386→            } else {
   387→                _ui.value = _ui.value.copy(messages = messages, loading = false)
   388→                hub.markPeerRead(_ui.value.peerId)
   389→            }
   390→        }
   391→    }
   392→
   393→    fun send(text: String) {
   394→        val peerId = _ui.value.peerId
   395→        if (text.isBlank() || peerId.isBlank()) return
   396→        viewModelScope.launch {
   397→            val sent = social.send(peerId, text.trim())
   398→            if (sent != null) fetch()
   399→            else _ui.value = _ui.value.copy(error = "Сообщение не отправлено")
   400→        }
   401→    }
   402→
   403→    override fun onCleared() {
   404→        pollJob?.cancel()
   405→        super.onCleared()
   406→    }
   407→}

   456→        if (query.length < 2) {
   457→            _ui.value = _ui.value.copy(found = emptyList(), searching = false)

   459→        }

   506→class UserProfileViewModel : ViewModel() {
   507→    data class ProfileUi(
   508→        val userId: String = "",
   509→        val user: com.mq1.player.data.api.UserDto = com.mq1.player.data.api.UserDto(),
   510→        val online: Boolean = false,
   511→        val lastSeen: String? = null,
   512→        val friendship: com.mq1.player.data.api.FriendshipState = com.mq1.player.data.api.FriendshipState(),
   513→        val loading: Boolean = true,
   514→        val error: String? = null,
   515→        val busy: Boolean = false,
   516→        val message: String? = null
   517→    )
   518→
   519→    private val social = ServiceLocator.socialRepository
   520→    private val hub = ServiceLocator.socialHub
   521→    private val _ui = MutableStateFlow(ProfileUi())
   522→    val ui: StateFlow<ProfileUi> = _ui
   523→
   524→    fun load(userId: String) {
   525→        if (_ui.value.userId == userId && !_ui.value.loading) return
   526→        _ui.value = ProfileUi(userId = userId)
   527→        fetch()
   528→    }
   529→
   530→    fun refresh() = fetch()
   531→
   532→    private fun fetch() {
   533→        val id = _ui.value.userId
   534→        if (id.isBlank()) return
   535→        viewModelScope.launch {
   536→            social.userProfile(id)
   537→                .onSuccess { p ->
   538→                    _ui.value = _ui.value.copy(
   539→                        user = p.user,
   540→                        online = p.online,
   541→                        lastSeen = p.lastSeen,
   542→                        friendship = p.friendship,
   543→                        loading = false,
   544→                        error = null
   545→                    )
   546→                }
   547→                .onFailure {
   548→                    _ui.value = _ui.value.copy(loading = false, error = it.message ?: "Профиль недоступен")
   549→                }
   550→        }
   551→    }
   552→
   553→    private fun runAction(action: suspend () -> Result<Unit>, onSuccess: String) {
   554→        if (_ui.value.busy) return
   555→        _ui.value = _ui.value.copy(busy = true)
   556→        viewModelScope.launch {
   557→            action()
   558→                .onSuccess {
   559→                    _ui.value = _ui.value.copy(message = onSuccess)
   560→                    fetch() // re-read server state (friendship status changed)

   624→ * F8 Lyrics — real data only: /api/music/lyrics (lrclib-backed, 10-min cache

   627→class LyricsViewModel : ViewModel() {
   628→
   629→    data class LyricsUi(
   630→        val trackKey: String = "",
   631→        val lines: List<com.mq1.player.data.api.LyricLine> = emptyList(),
   632→        val plainText: String = "",
   633→        val synced: Boolean = false,
   634→        val loading: Boolean = false,
   635→        /** true when the fetch completed but nothing was found. */
   636→        val unavailable: Boolean = false
   637→    )
   638→
   639→    private val music = ServiceLocator.musicRepository
   640→    private val _ui = MutableStateFlow(LyricsUi())
   641→    val ui: StateFlow<LyricsUi> = _ui
   642→
   643→    /** Load (once per track) when the lyrics panel opens. */
   644→    fun loadIfNeeded(track: com.mq1.player.data.api.Track) {
   645→        val key = track.scTrackId?.toString() ?: track.id
   646→        if (_ui.value.trackKey == key && (_ui.value.loading || _ui.value.synced ||
   647→                _ui.value.plainText.isNotBlank() || _ui.value.unavailable)) return
   648→        _ui.value = LyricsUi(trackKey = key, loading = true)
   649→        viewModelScope.launch {
   650→            val result = music.lyrics(track.artist, track.title)
   651→            _ui.value = if (result == null) {
   652→                _ui.value.copy(loading = false, unavailable = true)
   653→            } else if (result.synced && result.lyrics.isNotEmpty()) {
   654→                // Server-side LRC parse occasionally leaves an embedded
   655→                // [mm:ss.xx] prefix in the text — strip it for display.
   656→                val embeddedTs = Regex("^\\[\\d{2}:\\d{2}(?:[.:]\\d{2,3})?]\\s*")
   657→                _ui.value.copy(
   658→                    loading = false,
   659→                    lines = result.lyrics.map { it.copy(text = it.text.replace(embeddedTs, "")) },
   660→                    synced = true
   661→                )
   662→            } else if (result.plainText.isNotBlank()) {
   663→                _ui.value.copy(loading = false, plainText = result.plainText)
   664→            } else {
   665→                _ui.value.copy(loading = false, unavailable = true)
   666→            }
   667→        }
   668→    }
   669→}

   688→    data class ProfileUi(
   689→        // ── Account (server) ──

   712→        /** Top artists derived from likes — tap → artist screen. */

   714→            get() = likes.groupBy { it.artist }
   715→                .filterKeys { it.isNotBlank() }

   750→    fun refresh() {
   751→        viewModelScope.launch {
   752→            _ui.value = _ui.value.copy(loading = true, error = null)
   753→
   754→            val account = profileRepo.myProfile()
   755→            if (account.isFailure) {
   756→                _ui.value = _ui.value.copy(
   757→                    loading = false,
   758→                    error = account.exceptionOrNull()?.message ?: "Профиль недоступен"
   759→                )
   760→                return@launch
   761→            }
   762→            val me = runCatching {
   763→                ServiceLocator.api.me().body()
   764→            }.getOrNull()
   765→
   766→            val myPlaylists = runCatching { playlistsRepo.myPlaylists() }.getOrElse { emptyList() }
   767→
   768→            _ui.value = _ui.value.copy(
   769→                userId = account.getOrThrow().id,
   770→                username = account.getOrThrow().username,
   771→                email = account.getOrThrow().email,
   772→                avatar = account.getOrThrow().avatar,
   773→                role = account.getOrThrow().role,
   774→                createdAt = account.getOrThrow().createdAt,
   775→                telegramUsername = me?.telegramUsername,
   776→                confirmed = me?.confirmed,
   777→                playlists = myPlaylists,
   778→                loading = false
   779→            )
   780→        }
   781→    }

   787→    fun saveUsername(newUsername: String) {
   788→        if (_ui.value.savingUsername) return
   789→        val localError = com.mq1.player.data.repo.ProfileRepository.validateUsername(newUsername)
   790→        if (localError != null || newUsername == _ui.value.username) {
   791→            _ui.value = _ui.value.copy(message = localError ?: "Имя не изменилось")
   792→            return
   793→        }
   794→        viewModelScope.launch {
   795→            _ui.value = _ui.value.copy(savingUsername = true)
   796→            profileRepo.updateUsername(newUsername)
   797→                .onSuccess { response ->
   798→                    _ui.value = _ui.value.copy(
   799→                        username = response.username ?: newUsername,
   800→                        savingUsername = false,
   801→                        message = response.message ?: "Имя обновлено"
   802→                    )
   803→                    // Session cache → all surfaces (Settings, chats, hub gate).
   804→                    local.sessionUser.firstOrNull()?.let { cached ->
   805→                        local.setSessionUser(cached.copy(username = response.username ?: newUsername))
   806→                    }
   807→                }
   808→                .onFailure {
   809→                    _ui.value = _ui.value.copy(
   810→                        savingUsername = false,
   811→                        message = it.message ?: "Ошибка сохранения имени"
   812→                    )
   813→                }
   814→        }
   815→    }

   817→    fun saveAvatar(imageBytes: ByteArray) {
   818→        if (_ui.value.savingAvatar) return
   819→        viewModelScope.launch(kotlinx.coroutines.Dispatchers.IO) {
   820→            val dataUrl = com.mq1.player.data.repo.ProfileRepository.bitmapToAvatarDataUrl(imageBytes)
   821→            if (dataUrl == null) {
   822→                _ui.value = _ui.value.copy(message = "Не удалось прочитать изображение")
   823→                return@launch
   824→            }
   825→            _ui.value = _ui.value.copy(savingAvatar = true, avatar = dataUrl) // optimistic
   826→            profileRepo.uploadAvatar(dataUrl)
   827→                .onSuccess { response ->
   828→                    _ui.value = _ui.value.copy(
   829→                        avatar = response.avatar ?: dataUrl,
   830→                        savingAvatar = false,
   831→                        message = response.message ?: "Аватарка обновлена"
   832→                    )
   833→                    local.sessionUser.firstOrNull()?.let { cached ->
   834→                        local.setSessionUser(cached.copy(avatar = dataUrl))
   835→                    }
   836→                }
   837→                .onFailure {
   838→                    _ui.value = _ui.value.copy(
   839→                        // rollback to the server value we still have in profile
   840→                        avatar = runCatching {
   841→                            profileRepo.myProfile().getOrNull()?.avatar
   842→                        }.getOrNull() ?: _ui.value.avatar,
   843→                        savingAvatar = false,
   844→                        message = it.message ?: "Ошибка загрузки аватара"
   845→                    )
   846→                }
   847→        }
   848→    }
