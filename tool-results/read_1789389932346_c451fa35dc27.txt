   592→/**
   593→ * WEB PARITY slider — 4dp track (muted 35% base, text fill) + 12dp thumb,
   594→ * tap + drag scrubbing (FullTrackViewMobile ProgressBar contract).
   595→ */
   596→@Composable
   597→private fun WebProgressSlider(
   598→    fraction: Float,
   599→    onScrub: (Float) -> Unit,
   600→    onScrubEnd: () -> Unit,
   601→    modifier: Modifier = Modifier
   602→) {
   603→    val trackColor = MaterialTheme.colorScheme.onSurfaceVariant.copy(alpha = 0.35f)
   604→    val fillColor = MaterialTheme.colorScheme.onBackground
   605→    var widthPx by remember { mutableStateOf(0f) }
   606→    Box(
   607→        modifier = modifier
   608→            .fillMaxWidth()
   609→            .height(28.dp)
   610→            .onSizeChanged { widthPx = it.width.toFloat() }
   611→            .pointerInput(Unit) {
   612→                detectTapGestures { offset ->
   613→                    if (widthPx > 0) onScrub((offset.x / widthPx).coerceIn(0f, 1f))
   614→                    onScrubEnd()
   615→                }
   616→            }
   617→            .pointerInput(Unit) {
   618→                detectDragGestures(
   619→                    onDragStart = { offset ->
   620→                        if (widthPx > 0) onScrub((offset.x / widthPx).coerceIn(0f, 1f))
   621→                    },
   622→                    onDragEnd = { onScrubEnd() }
   623→                ) { change, _ ->
   624→                    if (widthPx > 0) onScrub((change.position.x / widthPx).coerceIn(0f, 1f))
   625→                }
   626→            }
   627→    ) {
   628→        // 4dp track
   629→        Box(
   630→            Modifier
   631→                .align(Alignment.Center)
   632→                .fillMaxWidth()
   633→                .height(4.dp)
   634→                .clip(RoundedCornerShape(2.dp))
   635→                .background(trackColor)
   636→        ) {
   637→            Box(
   638→                Modifier
   639→                    .fillMaxWidth(fraction)
   640→                    .height(4.dp)
   641→                    .background(fillColor)
   642→            )
   643→        }
   644→        // 12dp thumb at the fill edge
   645→        val density = androidx.compose.ui.platform.LocalDensity.current
   646→        Box(
   647→            Modifier
   648→                .align(Alignment.CenterStart)
   649→                .offset(x = with(density) { ((fraction * widthPx) - 6.dp.toPx()).coerceAtLeast(0f).toDp() })
   650→                .size(12.dp)
   651→                .clip(CircleShape)
   652→                .background(fillColor)
   653→        )
   654→    }
   655→}

   657→/** Synced lyrics: current line highlighted + auto-scrolled, tap line → seek. */
   658→@Composable
   659→private fun SyncedLyricsList(
   660→    lines: List<com.mq1.player.data.api.LyricLine>,
   661→    positionMs: Long,
   662→    onSeek: (Long) -> Unit
   663→) {
   664→    val listState = rememberLazyListState()
   665→    val positionSec = positionMs / 1000.0
   666→    val currentIndex = lines.indexOfLast { it.time <= positionSec }
   667→    val accent = MaterialTheme.colorScheme.primary
   668→    val muted = MaterialTheme.colorScheme.onSurfaceVariant
   669→
   670→    LaunchedEffect(currentIndex) {
   671→        if (currentIndex >= 0) {
   672→            runCatching { listState.animateScrollToItem(currentIndex) }
   673→        }
   674→    }
   675→
   676→    LazyColumn(
   677→        state = listState,
   678→        contentPadding = androidx.compose.foundation.layout.PaddingValues(
   679→            horizontal = 20.dp, vertical = 8.dp
   680→        ),
   681→        modifier = Modifier.height(440.dp)
   682→    ) {
   683→        itemsIndexed(lines, key = { i, _ -> i }) { i, line ->
   684→            val isCurrent = i == currentIndex
   685→            Text(
   686→                line.text.ifBlank { "♪" },
   687→                style = if (isCurrent) MqType.track.copy(fontSize = 16.sp) else MqType.body,
   688→                color = if (isCurrent) accent else muted,
   689→                textAlign = TextAlign.Start,
   690→                modifier = Modifier
   691→                    .fillMaxWidth()
   692→                    .clickable(enabled = line.text.isNotBlank()) { onSeek((line.time * 1000).toLong()) }
   693→                    .padding(vertical = 8.dp)
   694→            )
   695→        }
   696→    }
   697→}
