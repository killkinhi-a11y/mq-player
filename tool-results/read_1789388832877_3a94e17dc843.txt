    69→/**
    70→ * WEB PARITY library — exact port of LibraryView.tsx (375px) with its lazy
    71→ * sub-views (FavoritesView / PlaylistView grid / HistoryView):
    72→ *
    73→ *  ┌ header: «Библиотека» 24/700 + «N элементов в коллекции»
    74→ *  ├ underline tab switcher: Избранное / Плейлисты / История (icon + label
    75→ *  │ + count badge, accent underline, 48dp rows)
    76→ *  ├ toolbar: «Поиск в библиотеке...» r12 + «Недавние» sort chip
    77→ *  ├ FAVORITES: icon chip + «Избранное» 18/700 + «N понр. · N не понр. ·
    78→ *  │ N подписок», pill tabs (icon + count), action bar («Найти трек...» +
    79→ *  │ sort + shuffle), «Слушать все»/«Перемешать» pills, r16 list card /
    80→ *  │ «Пока пусто» empty (72 r24 heart chip + «Искать музыку» CTA)
    81→ *  ├ PLAYLISTS: header + «Создать»/«Импорт», 2-col tile grid (r16 cards,
    82→ *  │ gradient art + ListMusic), mq-empty «Нет плейлистов», inline create
    83→ *  │ card («Новый плейлист», Отмена/Создать)
    84→ *  └ HISTORY: 48 r16 clock chip + «История» 20/700 + «Слушать всё»/clear,
    85→ *     search bar, 3+2 stat cards, grouped day cards («Сегодня»…), or
    86→ *     «История пуста» hero (96 r24 clock chip + «Начать слушать» CTA)
    87→ *
    88→ * LibraryBody is stateless (fixture-renderable for parity screenshots);
    89→ * LibraryScreen wires the live ViewModels into it.
    90→ */
    91→@Composable
    92→fun LibraryScreen(onOpenPlaylist: (String) -> Unit) {
    93→    val vm: PlaylistViewModel = viewModel()
    94→    val player: PlayerViewModel = viewModel()
    95→    val ui by vm.ui.collectAsState()
    96→    val queue by player.controller.queue.collectAsState()
    97→    val currentIndex by player.controller.currentIndex.collectAsState()
    98→    val favorites by player.favorites.collectAsState(initial = emptyList())
    99→    val history by ServiceLocator.localStore.history.collectAsState(initial = emptyList())
   100→    val scope = androidx.compose.runtime.rememberCoroutineScope()
   101→
   102→    var tab by remember { mutableIntStateOf(0) }
   103→    var query by remember { mutableStateOf("") }
   104→    var sort by remember { mutableIntStateOf(0) }
   105→    var showCreate by remember { mutableStateOf(false) }
   106→    var newName by remember { mutableStateOf("") }
   107→
   108→    LaunchedEffect(Unit) { vm.refresh() }
   109→
   110→    LibraryBody(
   111→        ui = ui,
   112→        activeTab = tab,
   113→        favorites = favorites,
   114→        history = history,
   115→        playingTrackId = queue.getOrNull(currentIndex)?.id,
   116→        favoriteIds = favorites.map { it.id }.toSet(),
   117→        query = query,
   118→        sort = sort,
   119→        showCreate = showCreate,
   120→        createName = newName,
   121→        onTabChange = { tab = it; query = "" }, // web clears search on tab switch
   122→        onQueryChange = { query = it },
   123→        onSortChange = { sort = it },
   124→        onCreateToggle = { showCreate = !showCreate },
   125→        onCreateNameChange = { newName = it },
   126→        onCreateConfirm = { name ->
   127→            vm.create(name) { showCreate = false; newName = "" }
   128→        },
   129→        onOpenPlaylist = onOpenPlaylist,
   130→        onPlayQueue = { q, i -> player.controller.playQueue(q, i) },
   131→        onFavorite = { player.controller.toggleFavorite(it) },
   132→        onClearHistory = { scope.launch { ServiceLocator.localStore.clearHistory() } },
   133→        onGoHome = { },
   134→    )
   135→}

  1221→/** web hashString: h = char + (h << 5) - h (int32 wrap, abs). */
  1222→private fun hashString(s: String): Int {
  1223→    var h = 0
  1224→    for (c in s) h = c.code + ((h shl 5) - h)
  1225→    return kotlin.math.abs(h)
  1226→}
