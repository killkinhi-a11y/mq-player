    39→    39→function getSleepPhase(elapsedMinutes: number): SleepPhase {
    40→    40→  const pos = elapsedMinutes % SLEEP_CYCLE_MINUTES;
    41→    41→  for (const phase of SLEEP_PHASES) {
    42→    42→    if (pos >= phase.startMin && pos < phase.endMin) return phase;
    43→    43→  }
    44→    44→  return SLEEP_PHASES[0];

    76→    76→function ScrollPicker({
    77→    77→  options,
    78→    78→  selected,
    79→    79→  onSelect,
    80→    80→}: {
    81→    81→  options: number[];
    82→    82→  selected: number;
    83→    83→  onSelect: (v: number) => void;
    84→    84→}) {
    85→    85→  const scrollRef = useRef<HTMLDivElement>(null);
    86→    86→  const scrollTimeout = useRef<ReturnType<typeof setTimeout> | null>(null);
    87→    87→  const itemHeight = 48;
    88→    88→
    89→    89→  useEffect(() => {
    90→    90→    const idx = options.indexOf(selected);
    91→    91→    if (idx >= 0 && scrollRef.current) {
    92→    92→      const targetScroll = 72 + idx * itemHeight - (192 / 2 - itemHeight / 2);
    93→    93→      scrollRef.current.scrollTop = targetScroll;
    94→    94→    }
    95→    95→  }, [selected, options]);
    96→    96→
    97→    97→  const handleScroll = useCallback(() => {
    98→    98→    if (scrollTimeout.current) clearTimeout(scrollTimeout.current);
    99→    99→    scrollTimeout.current = setTimeout(() => {}, 150);
   100→   100→  }, []);
   101→   101→
   102→   102→  const handleScrollEnd = useCallback(() => {
   103→   103→    const el = scrollRef.current;
   104→   104→    if (!el) return;
   105→   105→    const scrollCenter = el.scrollTop + 192 / 2;
   106→   106→    const idx = Math.floor((scrollCenter - 72) / itemHeight);
   107→   107→    if (idx >= 0 && idx < options.length) {
   108→   108→      onSelect(options[idx]);
   109→   109→      const targetScroll = 72 + idx * itemHeight - (192 / 2 - itemHeight / 2);
   110→   110→      el.scrollTo({ top: targetScroll, behavior: "smooth" });
   111→   111→    }
   112→   112→  }, [options, onSelect]);
   113→   113→
   114→   114→  const getOpacity = useCallback(
   115→   115→    (idx: number) => {
   116→   116→      if (!scrollRef.current) return idx === options.indexOf(selected) ? 1 : 0.35;
   117→   117→      const scrollCenter = scrollRef.current.scrollTop + 192 / 2;
   118→   118→      const itemCenter = 72 + idx * itemHeight + itemHeight / 2;
   119→   119→      const distance = Math.abs(scrollCenter - itemCenter);
   120→   120→      if (distance < itemHeight / 2) return 1;
   121→   121→      if (distance > itemHeight * 2.5) return 0.15;
   122→   122→      return 0.35 + 0.65 * Math.max(0, 1 - distance / (itemHeight * 2.5));
   123→   123→    },
   124→   124→    [selected, options]
   125→   125→  );
   126→   126→
   127→   127→  const getScale = useCallback(
   128→   128→    (idx: number) => {
   129→   129→      if (!scrollRef.current) return idx === options.indexOf(selected) ? 1 : 0.9;
   130→   130→      const scrollCenter = scrollRef.current.scrollTop + 192 / 2;
   131→   131→      const itemCenter = 72 + idx * itemHeight + itemHeight / 2;
   132→   132→      const distance = Math.abs(scrollCenter - itemCenter);
   133→   133→      if (distance < itemHeight / 2) return 1;
   134→   134→      if (distance > itemHeight * 2) return 0.85;
   135→   135→      return 0.85 + 0.15 * Math.max(0, 1 - distance / (itemHeight * 2));
   136→   136→    },
   137→   137→    [selected, options]
   138→   138→  );
   139→   139→
   140→   140→  return (
   141→   141→    <div className="relative h-[192px] overflow-hidden rounded-2xl lg:hidden" style={{ backgroundColor: "var(--mq-card)" }}>
   142→   142→      <div className="absolute top-0 left-0 right-0 h-16 z-10 pointer-events-none rounded-t-2xl" style={{ background: "linear-gradient(var(--mq-card), transparent)" }} />
   143→   143→      <div className="absolute bottom-0 left-0 right-0 h-16 z-10 pointer-events-none rounded-b-2xl" style={{ background: "linear-gradient(transparent, var(--mq-card))" }} />
   144→   144→      <div className="absolute left-3 right-3 h-12 pointer-events-none z-[5] rounded-xl" style={{ backgroundColor: "var(--mq-accent)", opacity: 0.12, border: "1px solid var(--mq-accent)", top: "calc(50% - 24px)" }} />
   145→   145→      <div ref={scrollRef} onScroll={handleScroll} onTouchEnd={handleScrollEnd} onMouseUp={handleScrollEnd} className="h-full overflow-y-auto px-4" style={{ scrollbarWidth: "none", WebkitOverflowScrolling: "touch" }}>
   146→   146→        <div style={{ height: 72 }} />
   147→   147→        {options.map((val, idx) => (
   148→   148→          <div key={val} className="h-12 flex items-center justify-center cursor-pointer select-none" style={{ opacity: getOpacity(idx), transform: `scale(${getScale(idx)})`, transition: "opacity 0.15s ease, transform 0.15s ease" }} onClick={() => { onSelect(val); if (scrollRef.current) { const t = 72 + idx * itemHeight - (192 / 2 - itemHeight / 2); scrollRef.current.scrollTo({ top: t, behavior: "smooth" }); } }}>
   149→   149→            <span className="text-lg font-semibold tracking-wide" style={{ color: selected === val ? "var(--mq-accent)" : "var(--mq-text)" }}>{formatOption(val)}</span>
   150→   150→          </div>
   151→   151→        ))}
   152→   152→        <div style={{ height: 72 }} />
   153→   153→      </div>
   154→   154→    </div>
   155→   155→  );
   156→   156→}

   159→   159→function CircularTimer({ size = 280, progress, remainingSeconds, isRunning, isPaused }: { size?: number; progress: number; remainingSeconds: number; isRunning: boolean; isPaused: boolean }) {
   160→   160→  const radius = (size - 16) / 2;
   161→   161→  const circumference = 2 * Math.PI * radius;
   162→   162→  const strokeDashoffset = circumference * (1 - progress);
   163→   163→
   164→   164→  const hours = Math.floor(remainingSeconds / 3600);
   165→   165→  const minutes = Math.floor((remainingSeconds % 3600) / 60);
   166→   166→  const seconds = remainingSeconds % 60;
   167→   167→
   168→   168→  // Sleep phase
   169→   169→  const elapsedMinutes = progress > 0 ? (1 - progress) * (remainingSeconds / 60 + (1 - progress) * remainingSeconds / 60) : 0;
   170→   170→  const totalSetMinutes = remainingSeconds > 0 ? Math.round(remainingSeconds / (1 - progress + 0.001) / 60) : 0;
   171→   171→  const actualElapsed = totalSetMinutes > 0 ? totalSetMinutes * progress : 0;
   172→   172→  const phase = isRunning ? getSleepPhase(actualElapsed) : null;
   173→   173→  const currentCycle = totalSetMinutes > 0 ? Math.floor(actualElapsed / SLEEP_CYCLE_MINUTES) + 1 : 0;
   174→   174→
   175→   175→  return (
   176→   176→    <div className={`relative ${isRunning ? "timer-glow-pc" : ""}`}>
   177→   177→      <svg width={size} height={size} className="transform -rotate-90">
   178→   178→        {/* Background ring */}
   179→   179→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="var(--mq-border)" strokeWidth="6" opacity={0.3} />
   180→   180→        {/* Progress ring */}
   181→   181→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="url(#timerGradPc)" strokeWidth="6" strokeLinecap="round" strokeDasharray={circumference} strokeDashoffset={strokeDashoffset} className="transition-all duration-1000 ease-linear" />
   182→   182→        {/* Glow track */}
   183→   183→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="url(#timerGradPc)" strokeWidth="2" strokeLinecap="round" strokeDasharray={circumference} strokeDashoffset={strokeDashoffset} className="transition-all duration-1000 ease-linear" style={{ filter: "blur(6px)", opacity: 0.4 }} />
   184→   184→        <defs>
   185→   185→          <linearGradient id="timerGradPc" x1="0%" y1="0%" x2="100%" y2="100%">
   186→   186→            <stop offset="0%" stopColor="#8b5cf6" />
   187→   187→            <stop offset="50%" stopColor="#6366f1" />
   188→   188→            <stop offset="100%" stopColor="#06b6d4" />
   189→   189→          </linearGradient>
   190→   190→        </defs>
   191→   191→      </svg>
   192→   192→      {/* Center content */}
   193→   193→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   194→   194→        {isPaused ? (
   195→   195→          <div className="flex items-center gap-2 mb-1">
   196→   196→            <Pause className="w-4 h-4" style={{ color: "var(--mq-accent)" }} />
   197→   197→            <span className="text-xs font-medium" style={{ color: "var(--mq-accent)" }}>Пауза</span>
   198→   198→          </div>
   199→   199→        ) : null}
   200→   200→        <span className="font-bold tabular-nums tracking-wider" style={{ fontSize: hours > 0 ? "2.5rem" : "3.2rem", color: "var(--mq-text)", lineHeight: 1 }}>
   201→   201→          {hours > 0 && <span className="text-2xl font-light" style={{ color: "var(--mq-text-muted)" }}>{hours}:</span>}
   202→   202→          {minutes.toString().padStart(2, "0")}:{seconds.toString().padStart(2, "0")}
   203→   203→        </span>
   204→   204→        {phase && isRunning && (
   205→   205→          <motion.div key={phase.name} initial={{ opacity: 0, y: 4 }} animate={{ opacity: 1, y: 0 }} className="mt-2 text-center">
   206→   206→            <p className="mq-t-meta-2 font-medium" style={{ color: phase.color }}>Цикл {currentCycle} · {phase.label}</p>
   207→   207→            <div className="w-14 h-1 rounded-full mt-1 overflow-hidden" style={{ backgroundColor: "var(--mq-border)" }}>
   208→   208→              <div className="h-full rounded-full transition-all duration-1000" style={{ backgroundColor: phase.color, width: `${((phase.endMin / SLEEP_CYCLE_MINUTES) * 100)}%` }} />
   209→   209→            </div>
   210→   210→          </motion.div>
   211→   211→        )}
   212→   212→      </div>
   213→   213→    </div>
   214→   214→  );
   215→   215→}

   229→   229→// ── Main View ──
   230→   230→// Seeded PRNG (mulberry32) — module scope, pure factory: the same seed
   231→   231→// always yields the same starfield on server AND client (no hydration
   232→   232→// mismatch, no Math.random() during render).
   233→   233→function mulberry32(seed: number): () => number {
   234→   234→  let s = seed | 0;
   235→   235→  return () => {
   236→   236→    s = (s + 0x6d2b79f5) | 0;
   237→   237→    let t = Math.imul(s ^ (s >>> 15), 1 | s);
   238→   238→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
   239→   239→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   240→   240→  };
   241→   241→}

   453→   453→function SleepCycleBar({ progress, totalMinutes }: { progress: number; totalMinutes: number }) {
   454→   454→  const elapsed = totalMinutes * progress;
   455→   455→  const cycleNum = Math.floor(elapsed / SLEEP_CYCLE_MINUTES) + 1;
   456→   456→  const totalCycles = Math.ceil(totalMinutes / SLEEP_CYCLE_MINUTES);
   457→   457→  const phase = getSleepPhase(elapsed);
   458→   458→
   459→   459→  return (
   460→   460→    <div className="space-y-2">
   461→   461→      <div className="flex items-center justify-between">
   462→   462→        <span className="text-sm font-semibold" style={{ color: phase.color }}>{phase.label}</span>
   463→   463→        <span className="text-xs" style={{ color: "var(--mq-text-muted)" }}>Цикл {cycleNum} из {totalCycles}</span>
   464→   464→      </div>
   465→   465→      {/* Phase bar */}
   466→   466→      <div className="flex h-2 rounded-full overflow-hidden" style={{ backgroundColor: "var(--mq-border)", opacity: 0.4 }}>
   467→   467→        {SLEEP_PHASES.map((p) => {
   468→   468→          const widthPct = ((p.endMin - p.startMin) / SLEEP_CYCLE_MINUTES) * 100;
   469→   469→          const posInCycle = (elapsed % SLEEP_CYCLE_MINUTES);
   470→   470→          const isCurrentPhase = posInCycle >= p.startMin && posInCycle < p.endMin;
   471→   471→          return (
   472→   472→            <div key={p.name} className="h-full transition-opacity duration-500" style={{ width: `${widthPct}%`, backgroundColor: isCurrentPhase ? p.color : p.color, opacity: isCurrentPhase ? 1 : 0.25 }} />
   473→   473→          );
   474→   474→        })}
   475→   475→        {/* Position marker */}
   476→   476→        <div className="absolute h-3 w-0.5 rounded-full -translate-y-0.5 transition-all duration-1000" style={{ left: `${(elapsed % SLEEP_CYCLE_MINUTES) / SLEEP_CYCLE_MINUTES * 100}%`, backgroundColor: "#fff", top: "0", boxShadow: "0 0 6px rgba(255,255,255,0.5)", position: "relative" }} />
   477→   477→      </div>
   478→   478→    </div>
   479→   479→  );
   480→   480→}
