   252→@Composable
   253→private fun AuthLandingCard(
   254→    onEmail: () -> Unit,
   255→    onRegister: () -> Unit,
   256→    onLoggedIn: (LocalStore.SessionUser) -> Unit,
   257→    onDemo: (LocalStore.SessionUser) -> Unit
   258→) {
   259→    val context = LocalContext.current
   260→    val scope = rememberCoroutineScope()
   261→    val auth = ServiceLocator.authRepository
   262→
   263→    var botName by remember { mutableStateOf<String?>(null) }
   264→    var botConfigured by remember { mutableStateOf<Boolean?>(null) }
   265→    var code by remember { mutableStateOf("") }
   266→    var username by remember { mutableStateOf("") }
   267→    var needsUsername by remember { mutableStateOf(false) }
   268→    var error by remember { mutableStateOf<String?>(null) }
   269→    var busy by remember { mutableStateOf(false) }
   270→    var demoBusy by remember { mutableStateOf(false) }
   271→    var googleBusy by remember { mutableStateOf(false) }
   272→
   273→    /**
   274→     * REAL Google native login — the CURRENT official Credential Manager
   275→     * sequence, orchestrated by [com.mq1.player.data.repo.GoogleAuthFlow]:
   276→     *
   277→     *   pass 1: GetGoogleIdOption(filterByAuthorizedAccounts=true,
   278→     *           autoSelectEnabled=true, serverClientId=WEB_CLIENT_ID, nonce)
   279→     *   NoCredentialException → pass 2 with filterByAuthorizedAccounts=false
   280→     *           → the FULL Google account picker (every account on device)
   281→     *   → Google-issued id_token → POST /api/auth/google/native → the SAME
   282→     *   JWKS verification + account resolution as the web callback →
   283→     *   Set-Cookie session → SecureCookieJar → Home.
   284→     *
   285→     * The 2.3.0 bug: NoCredentialException ("no account previously authorized
   286→     * with THIS app" — the normal state for every first-time user) was shown
   287→     * as "Google-аккаунт не найден на устройстве" with no fallback pass, so
   288→     * first-time Google login was impossible. NoCredentialException on pass 1
   289→     * now falls through to the picker; only after BOTH passes can we honestly
   290→     * say there is no account.
   291→     *
   292→     * Honest failures only: every outcome has its own user message and a
   293→     * diagnostic log (provider/step/status/class — never token material).
   294→     * The button ALWAYS restarts the flow from scratch (fresh nonce per
   295→     * attempt; demo state is irrelevant to this path and is fully replaced
   296→     * by the authenticated state on success).
   297→     */
   298→    suspend fun googleNativeLogin() {
   299→        val outcome = com.mq1.player.data.repo.GoogleAuthFlow.login(
   300→            activityContext = context,
   301→            auth = auth
   302→        )
   303→        when (outcome) {
   304→            is com.mq1.player.data.repo.GoogleAuthFlow.GoogleAuthResult.CancelledByUser -> Unit
   305→            is com.mq1.player.data.repo.GoogleAuthFlow.GoogleAuthResult.Success ->
   306→                onLoggedIn(outcome.user)
   307→            is com.mq1.player.data.repo.GoogleAuthFlow.GoogleAuthResult.BackendRejected -> {
   308→                error = when (outcome.backendError) {
   309→                    "google_token_invalid" -> "Google-токен отклонён сервером"
   310→                    "invalid_nonce" -> "Код входа устарел. Попробуйте ещё раз"
   311→                    "blocked" -> "Аккаунт заблокирован"
   312→                    "maintenance" -> "Технические работы. Вход временно недоступен"
   313→                    "google_not_configured" -> "Google вход не настроен на сервере"
   314→                    else -> "Не удалось войти через Google"
   315→                }
   316→            }
   317→            // every remaining variant carries its own message
   318→            is com.mq1.player.data.repo.GoogleAuthFlow.GoogleAuthResult.NotConfigured ->
   319→                error = outcome.message
   320→            is com.mq1.player.data.repo.GoogleAuthFlow.GoogleAuthResult.NetworkError ->
   321→                error = outcome.message
   322→            is com.mq1.player.data.repo.GoogleAuthFlow.GoogleAuthResult.NoAccountPicked ->
   323→                error = outcome.message
   324→            is com.mq1.player.data.repo.GoogleAuthFlow.GoogleAuthResult.TokenParseFailed ->
   325→                error = outcome.message
   326→            is com.mq1.player.data.repo.GoogleAuthFlow.GoogleAuthResult.ProviderUnavailable ->
   327→                error = outcome.message
   328→            is com.mq1.player.data.repo.GoogleAuthFlow.GoogleAuthResult.CredentialManagerError ->
   329→                error = outcome.message
   330→            is com.mq1.player.data.repo.GoogleAuthFlow.GoogleAuthResult.BackendUnreachable ->
   331→                error = outcome.message
   332→            is com.mq1.player.data.repo.GoogleAuthFlow.GoogleAuthResult.TokenObtained -> Unit
   333→        }
   334→    }
   335→
   336→    LaunchedEffect(Unit) {
   337→        val name = auth.botName()
   338→        botConfigured = name != null
   339→        botName = name
   340→    }
   341→
   342→    AuthCard {
   343→        // header
   344→        Box(
   345→            modifier = Modifier
   346→                .size(64.dp)
   347→                .clip(RoundedCornerShape(16.dp))
   348→                .background(MaterialTheme.colorScheme.primary),
   349→            contentAlignment = Alignment.Center
   350→        ) {
   351→            Text("mq", style = MqType.display.copy(fontSize = 24.sp), color = Color.White)
   352→        }
   353→        Spacer(Modifier.height(16.dp))
   354→        Text(
   355→            "Вход в MQ Player",
   356→            style = MqType.page.copy(fontSize = 20.sp, fontWeight = FontWeight.W600),
   357→            color = MaterialTheme.colorScheme.onBackground,
   358→            modifier = Modifier.fillMaxWidth(), textAlign = TextAlign.Center
   359→        )
   360→        Text(
   361→            "Музыка, плейлисты и чаты синхронизированы между устройствами",
   362→            style = MqType.body.copy(fontSize = 14.sp, fontWeight = FontWeight.W400),
   363→            color = MaterialTheme.colorScheme.onSurfaceVariant,
   364→            modifier = Modifier.fillMaxWidth(), textAlign = TextAlign.Center
   365→        )
   366→        Spacer(Modifier.height(24.dp))
   367→
   368→        // Google — native Credential Manager login (the SAME backend flow the
   369→        // web finishes at /api/auth/google/callback, native transport)
   370→        WebButton(
   371→            text = "Продолжить с Google",
   372→            bg = Color.White,
   373→            border = Color(0xFFE5E7EB),
   374→            textColor = Color(0xFF1F2937),
   375→            icon = { GoogleLogo() },
   376→            enabled = !googleBusy && !busy,
   377→            loading = googleBusy
   378→        ) {
   379→            scope.launch {
   380→                googleBusy = true
   381→                error = null
   382→                try {
   383→                    googleNativeLogin()
   384→                } finally {
   385→                    googleBusy = false
   386→                }
   387→            }
   388→        }
   389→
   390→        // Telegram section
   391→        Spacer(Modifier.height(16.dp))
   392→        Box(Modifier.fillMaxWidth().height(1.dp).background(MaterialTheme.colorScheme.outline.copy(alpha = 0.22f)))
   393→        Spacer(Modifier.height(16.dp))
   394→
   395→        Row(horizontalArrangement = Arrangement.Center, verticalAlignment = Alignment.CenterVertically) {
   396→            MqIcon(icon = MqIcons.Send, size = 16.dp, tint = Telegram)
   397→            Spacer(Modifier.width(8.dp))
   398→            Text("Вход через Telegram", style = MqType.body.copy(fontSize = 14.sp, fontWeight = FontWeight.W500), color = MaterialTheme.colorScheme.onBackground)
   399→        }
   400→        Spacer(Modifier.height(12.dp))
   401→        Text(
   402→            "Откройте бота в Telegram и отправьте любое сообщение. Бот пришлёт 6-значный код подтверждения",
   403→            style = MqType.meta.copy(fontSize = 12.sp),
   404→            color = MaterialTheme.colorScheme.onSurfaceVariant,
   405→            modifier = Modifier.fillMaxWidth(), textAlign = TextAlign.Center
   406→        )
   407→        Spacer(Modifier.height(12.dp))
   408→
   409→        if (botConfigured == true && botName != null) {
   410→            WebButton(
   411→                text = "Открыть бота в Telegram",
   412→                bg = Telegram.copy(alpha = 0.15f),
   413→                border = Telegram.copy(alpha = 0.3f),
   414→                textColor = Telegram,
   415→                icon = { MqIcon(icon = MqIcons.Send, size = 16.dp, tint = Telegram) }
   416→            ) {
   417→                runCatching {
   418→                    context.startActivity(
   419→                        Intent(Intent.ACTION_VIEW, android.net.Uri.parse("https://t.me/$botName?start=code"))
   420→                    )
   421→                }
   422→            }
   423→            Spacer(Modifier.height(16.dp))
   424→        }
   425→
   426→        OtpBoxes(
   427→            code = code,
   428→            error = error != null,
   429→            onCode = { code = it }
   430→        )
   431→        if (needsUsername) {
   432→            Spacer(Modifier.height(12.dp))
   433→            WebInput(
   434→                value = username,
   435→                onValue = { if (it.length <= 20) username = it },
   436→                placeholder = "Имя пользователя (латиница)"
   437→            )
   438→        }
   439→        Spacer(Modifier.height(16.dp))
   440→
   441→        error?.let {
   442→            Text(
   443→                it,
   444→                style = MqType.body.copy(fontSize = 14.sp),
   445→                color = Color(0xFFFF6B6B),
   446→                modifier = Modifier.fillMaxWidth().padding(bottom = 12.dp),
   447→                textAlign = TextAlign.Center
   448→            )
   449→        }
   450→
   451→        WebButton(
   452→            text = "Подтвердить",
   453→            bg = Telegram,
   454→            border = Telegram,
   455→            textColor = Color.White,
   456→            enabled = code.length == 6 && !busy,
   457→            loading = busy
   458→        ) {
   459→            if (code.length != 6 || (needsUsername && username.isBlank())) return@WebButton
   460→            busy = true
   461→            error = null
   462→            scope.launch {
   463→                val result = auth.verifyCode(code, username.takeIf { needsUsername })
   464→                busy = false
   465→                when {
   466→                    result == null -> error = "Сеть недоступна. Проверьте подключение."
   467→                    result.error != null -> error = result.error
   468→                    result.isNewUser && !needsUsername -> {
   469→                        needsUsername = true
   470→                        error = "Новый пользователь — выберите имя"
   471→                    }
   472→                    result.userId != null -> onLoggedIn(
   473→                        LocalStore.SessionUser(
   474→                            userId = result.userId ?: "",
   475→                            username = result.username ?: username,
   476→                            role = result.role ?: "user",
   477→                            avatar = result.avatar

   551→/** 6 single-digit boxes — web 44×56 r12, filled border telegram / error red. */
   552→@Composable
   553→private fun OtpBoxes(code: String, error: Boolean, onCode: (String) -> Unit) {
   554→    val inputBg = LocalMqPalette.current.inputBg
   555→    val outline = MaterialTheme.colorScheme.outline
   556→    Row(horizontalArrangement = Arrangement.spacedBy(10.dp)) {
   557→        repeat(6) { i ->
   558→            val ch = code.getOrNull(i)?.toString() ?: ""
   559→            val borderColor = when {
   560→                error -> Color(0xFFEF4444)
   561→                ch.isNotEmpty() -> Telegram
   562→                else -> outline
   563→            }
   564→            BasicTextField(
   565→                value = ch,
   566→                onValueChange = { v ->
   567→                    if (v.length <= 1 && (v.isEmpty() || v[0].isDigit())) {
   568→                        onCode(code.take(i) + v + code.drop(i + 1))
   569→                    }
   570→                },
   571→                keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.NumberPassword),
   572→                singleLine = true,
   573→                textStyle = MqType.display.copy(
   574→                    fontSize = 20.sp, fontWeight = FontWeight.W700,
   575→                    textAlign = TextAlign.Center
   576→                ).let { it.copy(color = MaterialTheme.colorScheme.onBackground) },
   577→                modifier = Modifier
   578→                    .width(44.dp)
   579→                    .height(56.dp)
   580→                    .clip(RoundedCornerShape(12.dp))
   581→                    .background(inputBg)
   582→                    .border(2.dp, borderColor, RoundedCornerShape(12.dp))
   583→            )
   584→        }
   585→    }
   586→}

   588→/** Web-style form input (inputBg, hairline border, r12, 44dp). */
   589→@Composable
   590→internal fun WebInput(
   591→    value: String,
   592→    onValue: (String) -> Unit,
   593→    placeholder: String,
   594→    password: Boolean = false,
   595→    keyboardType: KeyboardType = KeyboardType.Text
   596→) {

   599→        onValueChange = onValue,
   600→        singleLine = true,
   601→        visualTransformation = if (password) PasswordVisualTransformation() else androidx.compose.ui.text.input.VisualTransformation.None,
   602→        keyboardOptions = KeyboardOptions(keyboardType = keyboardType),
   603→        textStyle = MqType.body.copy(color = MaterialTheme.colorScheme.onBackground),
   604→        decorationBox = { inner ->
   605→            Box(
   606→                Modifier
   607→                    .fillMaxWidth()
   608→                    .height(44.dp)
   609→                    .clip(RoundedCornerShape(12.dp))
   610→                    .background(LocalMqPalette.current.inputBg)
   611→                    .border(1.dp, MaterialTheme.colorScheme.outline, RoundedCornerShape(12.dp))
   612→                    .padding(horizontal = 14.dp),
   613→                contentAlignment = Alignment.CenterStart
   614→            ) {
   615→                if (value.isEmpty()) {
   616→                    Text(placeholder, style = MqType.body.copy(fontSize = 14.sp), color = MaterialTheme.colorScheme.onSurfaceVariant)
   617→                }
   618→                inner()
   619→            }
   620→        }
   621→    )

   626→@Composable
   627→private fun EmailLoginCard(
   628→    onBack: () -> Unit,
   629→    onLoggedIn: (LocalStore.SessionUser) -> Unit,
   630→    onNeedConfirm: () -> Unit
   631→) {
   632→    val scope = rememberCoroutineScope()

   688→private fun RegisterCard(
   689→    email: String,
   690→    onEmail: (String) -> Unit,
   691→    onBack: () -> Unit,
   692→    onRegistered: () -> Unit
   693→) {
   694→    val scope = rememberCoroutineScope()
   695→    val auth = ServiceLocator.authRepository
   696→    var username by remember { mutableStateOf("") }
   697→    var email by remember { mutableStateOf("") }
   698→    var password by remember { mutableStateOf("") }
   699→    var error by remember { mutableStateOf<String?>(null) }
   700→    var busy by remember { mutableStateOf(false) }

   706→        Spacer(Modifier.height(12.dp))

   708→        Spacer(Modifier.height(12.dp))
   709→        WebInput(password, { password = it }, "Пароль (мин. 6 символов)", password = true)
   710→        error?.let {
   711→            Spacer(Modifier.height(12.dp))
   712→            Text(it, style = MqType.body, color = Color(0xFFFF6B6B), modifier = Modifier.fillMaxWidth(), textAlign = TextAlign.Center)
   713→        }
   714→        Spacer(Modifier.height(16.dp))
   715→        WebButton(
   716→            text = "Создать аккаунт", bg = MaterialTheme.colorScheme.primary,
   717→            border = MaterialTheme.colorScheme.primary, textColor = MaterialTheme.colorScheme.onPrimary,
   718→            enabled = username.isNotBlank() && email.isNotBlank() && password.length >= 6 && !busy,
   719→            loading = busy
   720→        ) {
   721→            busy = true; error = null
   722→            scope.launch {
   723→                val result = auth.registerEmail(username, email, password)
   724→                busy = false
   725→                if (result == null) {
   726→                    error = "Ошибка соединения. Проверьте интернет."
   727→                } else if (result.devCode != null || result.error == null) {
   728→                    onRegistered()
   729→                } else {
   730→                    error = result.error ?: "Ошибка регистрации"
   731→                }
   732→            }
   733→        }
   734→    }
