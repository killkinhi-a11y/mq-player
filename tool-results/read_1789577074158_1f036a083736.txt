    41→function SyncedLyrics({ lines, onSeek }: { lines: SyncedLine[]; onSeek: (t: number) => void }) {
    42→  const activeRef = useRef<HTMLButtonElement | null>(null);
    43→  const progress = useAppStore((s) => s.progress);
    44→  const activeIdx = (() => {
    45→    let idx = -1;
    46→    for (let i = 0; i < lines.length; i++) { if (lines[i].time <= progress) idx = i; }
    47→    return idx;
    48→  })();
    49→  useEffect(() => {
    50→    activeRef.current?.scrollIntoView({ block: "center", behavior: "smooth" });
    51→  }, [activeIdx]);
    52→  return (
    53→    <div className="text-base leading-relaxed max-h-full overflow-y-auto px-2 py-2 space-y-1 scroll-smooth">
    54→      {lines.map((l, i) => (
    55→        <button
    56→          key={i}
    57→          ref={i === activeIdx ? activeRef : null}
    58→          onClick={() => onSeek(l.time)}
    59→          className="block w-full text-left px-2 py-1.5 rounded-lg transition-all duration-300 cursor-pointer"
    60→          style={{
    61→            color: i === activeIdx ? "var(--mq-text)" : "var(--mq-text-muted)",
    62→            opacity: i === activeIdx ? 1 : 0.62,
    63→            transform: i === activeIdx ? "scale(1)" : "scale(0.985)",
    64→            fontWeight: i === activeIdx ? 600 : 400,
    65→            background: "transparent",
    66→            border: "none",
    67→          }}
    68→        >
    69→          {l.text}
    70→        </button>
    71→      ))}
    72→    </div>
    73→  );
    74→}

    76→function EscapeHandler({ active, onEscape }: { active: boolean; onEscape: () => void }) {
    77→  useEffect(() => {
    78→    if (!active) return;
    79→    const h = (e: KeyboardEvent) => { if (e.key === "Escape") onEscape(); };
    80→    window.addEventListener("keydown", h);
    81→    return () => window.removeEventListener("keydown", h);
    82→  }, [active, onEscape]);
    83→  return null;
    84→}
