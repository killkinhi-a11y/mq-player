    38→function formatTime(iso: string): string {
    39→  try {
    40→    const d = new Date(iso);
    41→    if (isNaN(d.getTime())) return "";
    42→    return d.toLocaleTimeString("ru-RU", { hour: "2-digit", minute: "2-digit" });
    43→  } catch { return ""; }
    44→}
    45→
    46→function formatLastSeen(iso: string | null): string {
    47→  if (!iso) return "был(а) недавно";
    48→  try {
    49→    const d = new Date(iso);
    50→    if (isNaN(d.getTime())) return "был(а) недавно";
    51→    const diffMin = Math.floor((Date.now() - d.getTime()) / 60000);
    52→    if (diffMin < 1) return "только что";
    53→    if (diffMin < 60) return `был(а) ${diffMin} мин назад`;
    54→    const diffH = Math.floor(diffMin / 60);
    55→    if (diffH < 24) return `был(а) ${diffH} ч назад`;
    56→    return `был(а) ${d.toLocaleDateString("ru-RU")}`;
    57→  } catch { return "был(а) недавно"; }
    58→}

    64→function getDateLabel(iso: string): string {
    65→  try {
    66→    const d = new Date(iso);
    67→    if (isNaN(d.getTime())) return "";
    68→    const today = new Date(); today.setHours(0, 0, 0, 0);
    69→    const yesterday = new Date(today); yesterday.setDate(today.getDate() - 1);
    70→    const msgDay = new Date(d); msgDay.setHours(0, 0, 0, 0);
    71→    if (msgDay.getTime() === today.getTime()) return "Сегодня";
    72→    if (msgDay.getTime() === yesterday.getTime()) return "Вчера";
    73→    return d.toLocaleDateString("ru-RU", { day: "numeric", month: "long" });
    74→  } catch { return ""; }
    75→}

    95→function colorForId(id: string): string {
    96→  let hash = 0;
    97→  const s = id || "x";
    98→  for (let i = 0; i < s.length; i++) hash = (hash * 31 + s.charCodeAt(i)) | 0;
    99→  return AVATAR_COLORS[Math.abs(hash) % AVATAR_COLORS.length];

   123→function VoiceMessageBubble({ voiceUrl, duration, isMine }: {
   124→  voiceUrl: string; duration: number; isMine: boolean;
   125→}) {
   126→  const audioRef = useRef<HTMLAudioElement | null>(null);
   127→  const [playing, setPlaying] = useState(false);
   128→  const [currentTime, setCurrentTime] = useState(0);
   129→  const [dur, setDur] = useState(duration || 0);
   130→
   131→  useEffect(() => {
   132→    let cancelled = false;
   133→    try {
   134→      const audio = new Audio(voiceUrl);
   135→      audio.onloadedmetadata = () => { if (!cancelled) setDur(audio.duration || duration || 0); };
   136→      audio.ontimeupdate = () => { if (!cancelled) setCurrentTime(audio.currentTime || 0); };
   137→      audio.onended = () => { if (!cancelled) { setPlaying(false); setCurrentTime(0); } };
   138→      audioRef.current = audio;
   139→    } catch {}
   140→    return () => {
   141→      cancelled = true;
   142→      try { audioRef.current?.pause(); } catch {}
   143→      audioRef.current = null;
   144→    };
   145→  }, [voiceUrl, duration]);
   146→
   147→  const toggle = useCallback(() => {
   148→    try {
   149→      const a = audioRef.current;
   150→      if (!a) return;
   151→      if (playing) { a.pause(); setPlaying(false); }
   152→      else { a.play().catch(() => {}); setPlaying(true); }
   153→    } catch {}
   154→  }, [playing]);
   155→
   156→  const progress = dur > 0 ? (currentTime / dur) * 100 : 0;
   157→  // Deterministic waveform (no Math.random to avoid re-renders).
   158→  const bars = useMemo(() => {
   159→    const arr: number[] = [];
   160→    for (let i = 0; i < 28; i++) arr.push(0.25 + ((i * 7) % 10) / 12);
   161→    return arr;
   162→  }, []);
   163→
   164→  return (
   165→    <div className="flex items-center gap-2 min-w-[180px] py-0.5">
   166→      <motion.button onClick={toggle}
   167→        className="w-9 h-9 rounded-full flex items-center justify-center flex-shrink-0"
   168→        style={{ backgroundColor: isMine ? "color-mix(in srgb, var(--mq-text) 22%, transparent)" : "var(--mq-accent)", color: "#fff" }}
   169→        aria-label={playing ? "Пауза" : "Воспроизвести"}>
   170→        {playing ? <span className="mq-t-meta-2 leading-none">❚❚</span>
   171→          : <span className="text-xs leading-none ml-0.5">▶</span>}
   172→      </motion.button>
   173→      <div className="flex-1">
   174→        <div className="flex items-end gap-[2px] h-7">
   175→          {bars.map((h, i) => {
   176→            const isActive = (i / bars.length) * 100 < progress;
   177→            return (
   178→              <div key={i} className="w-[2px] rounded-full transition-colors"
   179→                style={{
   180→                  height: `${h * 100}%`,
   181→                  backgroundColor: isMine
   182→                    ? (isActive ? "rgba(255,255,255,0.95)" : "rgba(255,255,255,0.45)")
   183→                    : (isActive ? "var(--mq-accent)" : "var(--mq-text-muted)"),
   184→                }} />
   185→            );
   186→          })}
   187→        </div>
   188→        <div className="mq-t-meta-2 mt-1"
   189→          style={{ color: isMine ? "rgba(255,255,255,0.75)" : "var(--mq-text-muted)" }}>
   190→          {formatDuration(playing ? currentTime : dur)}
   191→        </div>
   192→      </div>
   193→    </div>
   194→  );
   195→}
