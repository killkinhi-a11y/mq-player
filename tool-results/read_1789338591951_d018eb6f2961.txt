    80→/**
    81→ * F9 — OWN profile (full, not just an avatar screen). Real data only:
    82→ * account from /api/user/profile + /api/auth/me, playlists from the backend,
    83→ * likes/history from the local store (web parity), friends from SocialHub.
    84→ * Transitions out: chat, artist, playlist, full player (track), friends,
    85→ * settings — all preserve this screen's ViewModel state on the back stack.
    86→ */
    87→@Composable
    88→fun MyProfileScreen(
    89→    onBack: () -> Unit,
    90→    onOpenChat: (peerId: String, peerName: String) -> Unit,
    91→    onOpenArtist: (String) -> Unit,
    92→    onOpenPlaylist: (id: String) -> Unit,
    93→    onOpenFriends: () -> Unit,
    94→    onOpenSettings: () -> Unit,
    95→    onOpenFullPlayer: () -> Unit,
    96→    onLogout: () -> Unit,
    97→    playTracks: (startIndex: Int) -> Unit
    98→) {
    99→    val vm: MyProfileViewModel = viewModel()
   100→    val ui by vm.ui.collectAsState()
   101→    val snackbar = remember { SnackbarHostState() }
   102→    val context = LocalContext.current
   103→    val scope = rememberCoroutineScope()
   104→
   105→    var editOpen by remember { mutableStateOf(false) }
   106→
   107→    // Photo picker → bytes → VM (decode/crop/encode on IO inside the VM path)
   108→    val imagePicker = rememberLauncherForActivityResult(
   109→        ActivityResultContracts.PickVisualMedia()
   110→    ) { uri ->
   111→        if (uri != null) {
   112→            scope.launch {
   113→                val bytes = withContext(Dispatchers.IO) {
   114→                    runCatching {
   115→                        context.contentResolver.openInputStream(uri)?.use { it.readBytes() }
   116→                    }.getOrNull()
   117→                }
   118→                if (bytes != null) vm.saveAvatar(bytes)
   119→            }
   120→        }
   121→    }
   122→
   123→    LaunchedEffect(ui.message) {
   124→        ui.message?.let {
   125→            snackbar.showSnackbar(it)
   126→            vm.consumeMessage()
   127→        }
   128→    }
   129→
   130→    Box(Modifier.fillMaxSize()) {
   131→        ProfileBody(
   132→            ui = ui,
   133→            onBack = onBack,
   134→            onRefresh = { vm.refresh() },
   135→            onChangeAvatar = {
   136→                imagePicker.launch(
   137→                    PickVisualMediaRequest(ActivityResultContracts.PickVisualMedia.ImageOnly)
   138→                )
   139→            },
   140→            onEdit = { editOpen = true },
   141→            onOpenChat = onOpenChat,
   142→            onOpenArtist = onOpenArtist,
   143→            onOpenPlaylist = onOpenPlaylist,
   144→            onOpenFriends = onOpenFriends,
   145→            onOpenSettings = onOpenSettings,
   146→            onOpenFullPlayer = onOpenFullPlayer,
   147→            onLogout = onLogout,
   148→            playTracks = playTracks
   149→        )
   150→
   151→        SnackbarHost(
   152→            hostState = snackbar,
   153→            modifier = Modifier.align(Alignment.BottomCenter)
   154→        )
   155→    }
   156→
   157→    if (editOpen) {
   158→        EditUsernameDialog(
   159→            current = ui.username,
   160→            saving = ui.savingUsername,
   161→            onCheck = { name -> vm.checkUsername(name) },
   162→            onSave = {
   163→                vm.saveUsername(it)
   164→                editOpen = false
   165→            },
   166→            onDismiss = { editOpen = false }
   167→        )
   168→    }
   169→}

   572→/** "Участник с …" from createdAt (ISO). */
   573→internal fun memberSince(iso: String?): String? {
   574→    val t = runCatching { java.time.Instant.parse(iso) }.getOrNull() ?: return null
   575→    // "d MMMM" (format pattern) → genitive month in Russian: "15 января 2026"
   576→    return java.time.format.DateTimeFormatter.ofPattern("d MMMM yyyy")
   577→        .withLocale(java.util.Locale("ru"))
   578→        .format(java.time.ZonedDateTime.ofInstant(t, java.time.ZoneId.systemDefault()))
   579→}

   659→/** Server availability verdict for the username being checked. */
   660→private data class CheckVerdict(val available: Boolean, val error: String?)
   661→
   662→/**
   663→ * Username edit dialog — same contract as the web: local validation (regex,
   664→ * length, reserved), debounced server availability check, then save.
   665→ */
   666→@Composable
   667→private fun EditUsernameDialog(
   668→    current: String,
   669→    saving: Boolean,
   670→    onCheck: suspend (String) -> UsernameCheckResponse?,
   671→    onSave: (String) -> Unit,
   672→    onDismiss: () -> Unit
   673→) {
   674→    var name by remember { mutableStateOf(current) }
   675→    // null | "checking" | CheckVerdict — network failure leaves no verdict
   676→    // (null) so the user can still attempt the save server-side.
   677→    var status by remember { mutableStateOf<Any?>(null) }
   678→
   679→    val localError = ProfileRepository.validateUsername(name)
   680→
   681→    // Debounced availability check (500ms — same as web)
   682→    LaunchedEffect(name) {
   683→        if (localError != null || name == current) {
   684→            status = null
   685→            return@LaunchedEffect
   686→        }
   687→        delay(500)
   688→        status = "checking"
   689→        val resp = onCheck(name)
   690→        status = resp?.let { CheckVerdict(it.available, it.error) }
   691→    }
   692→
   693→    AlertDialog(
   694→        onDismissRequest = if (saving) ({}) else onDismiss,
   695→        title = { Text("Имя пользователя") },
   696→        text = {
   697→            Column {
   698→                OutlinedTextField(
   699→                    value = name,
   700→                    onValueChange = { if (it.length <= 20) name = it },
   701→                    singleLine = true,
   702→                    label = { Text("Имя (2-20 символов)") },
   703→                    isError = localError != null || (status as? CheckVerdict)?.available == false,
   704→                    supportingText = {
   705→                        val msg = localError ?: when (val st = status) {
   706→                            "checking" -> "Проверяем…"
   707→                            is CheckVerdict -> if (st.available) "Имя свободно"
   708→                            else st.error ?: "Имя занято"
   709→                            else -> null
   710→                        }
   711→                        if (msg != null) Text(msg)
   712→                    }
   713→                )
   714→            }
   715→        },
   716→        confirmButton = {
   717→            Button(
   718→                onClick = { onSave(name.trim()) },
   719→                enabled = !saving && localError == null &&
   720→                    name != current && (status as? CheckVerdict)?.available != false
   721→            ) {
   722→                if (saving) {
   723→                    CircularProgressIndicator(
   724→                        strokeWidth = 2.dp, modifier = Modifier.size(18.dp)
   725→                    )
   726→                } else {
   727→                    Text("Сохранить")
   728→                }
   729→            }
   730→        },
   731→        dismissButton = {
   732→            TextButton(onClick = onDismiss, enabled = !saving) { Text("Отмена") }
   733→        }
   734→    )
   735→}
