    59→/**
    60→ * WEB PARITY search — exact port of SearchView.tsx mobile (375px) composition:
    61→ *
    62→ *  ┌ header: «Поиск» mq-t-display 26/800
    63→ *  ├ search bar: h48 r14 surface bg + hairline border, Search 18 left,
    64→ *  │ placeholder «Искать треки, артисты, альбомы...», X clear 28 circle
    65→ *  ├ mobile action row (right-aligned): «Фильтры» + «Файлы» h44 r12
    66→ *  ├ genre chips (collapsible): «Все» + 8 genres, pill r-full 12/600
    67→ *  ├ recent queries: «Недавние запросы» + «Очистить», clock chips r12
    68→ *  ├ results head: «Результаты» 20/700 + count + «Играть все» accent pill
    69→ *  ├ loading: 5 skeleton rows (art 44 + 2 lines + duration)
    70→ *  ├ no-results: mq-empty dashed card + 4 trending retry chips
    71→ *  ├ results: shared TrackRow list
    72→ *  └ idle hero: 88×88 r28 gradient Headphones chip, «Что послушаем?»,
    73→ *     «Популярные запросы» + 7 trending chips (Поп…Джаз)
    74→ *
    75→ * SearchBody is stateless (fixture-renderable for parity screenshots);
    76→ * SearchScreen wires the live ViewModels into it.
    77→ */
    78→@Composable
    79→fun SearchScreen(onOpenArtist: (String) -> Unit) {
    80→    val vm: SearchViewModel = viewModel()
    81→    val player: PlayerViewModel = viewModel()
    82→    val ui by vm.ui.collectAsState()
    83→    val queue by player.controller.queue.collectAsState()
    84→    val currentIndex by player.controller.currentIndex.collectAsState()
    85→    val favorites by player.favorites.collectAsState(initial = emptyList())
    86→
    87→    SearchBody(
    88→        ui = ui,
    89→        playingTrackId = queue.getOrNull(currentIndex)?.id,
    90→        favorites = favorites,
    91→        onQueryChange = vm::onQueryChange,
    92→        onClearHistory = vm::clearHistory,
    93→        onRemoveHistoryItem = vm::removeHistoryItem,
    94→        onPlayQueue = { q, i -> player.controller.playQueue(q, i) },
    95→        onFavorite = { player.controller.toggleFavorite(it) },
    96→        onOpenArtist = onOpenArtist,
    97→        onRetry = { vm.onQueryChange(ui.query) },
    98→    )
    99→}

   529→        }
   530→
   531→        // ── error state ─────────────────────────────────────────────────
   532→        if (ui.error != null) {
   533→            item { ErrorState(ui.error!!, onRetry = onRetry) }
   534→        }
   535→
   536→        // ── idle hero (web default empty state) ─────────────────────────
   537→        if (!ui.searched && !ui.loading && ui.error == null && ui.history.isEmpty()) {
   538→            item { SearchIdleHero(onQueryChange = onQueryChange) }
   539→        }
   540→
   541→        // ── trending chips (web shows these when history exists) ────────
   542→        if (!ui.searched && !ui.loading && ui.error == null && ui.history.isNotEmpty()) {
   543→            item { TrendingSection(onQueryChange = onQueryChange) }
   544→        }
   545→    }
   546→}
